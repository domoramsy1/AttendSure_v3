import ipaddress
import logging
import os
from django.db import transaction
from rest_framework import viewsets

logger = logging.getLogger(__name__)

TRUSTED_PROXIES = {
    ip.strip()
    for ip in os.getenv("TRUSTED_PROXIES", "127.0.0.1,::1").split(",")
    if ip.strip()
}


def _valid_ip(candidate: str | None) -> str | None:
    if not candidate:
        return None
    try:
        return str(ipaddress.ip_address(candidate.strip()))
    except ValueError:
        return None


def get_client_ip(request) -> str:
    if not request:
        return "127.0.0.1"
    remote = _valid_ip(request.META.get("REMOTE_ADDR")) or "127.0.0.1"
    if remote in TRUSTED_PROXIES:
        cf_ip = _valid_ip(request.META.get("HTTP_CF_CONNECTING_IP"))
        if cf_ip:
            return cf_ip
        xff = request.META.get("HTTP_X_FORWARDED_FOR")
        if xff:
            forwarded = _valid_ip(xff.split(",")[0])
            if forwarded:
                return forwarded
    return remote


def record_audit_log(
    *,
    request,
    table_name: str,
    record_id: int,
    action: str,
    changed_fields: list[str] | None = None,
    new_data: dict | None = None,
):
    from apps.accounts.models import AuditLog

    user = request.user if (request and request.user and request.user.is_authenticated) else None
    clean_fields = [str(field) for field in changed_fields] if changed_fields else None
    payload = dict(new_data or {})

    try:
        with transaction.atomic():
            AuditLog.objects.create(
                table_name=str(table_name)[:64],
                record_id=record_id or 0,
                action=str(action).upper()[:10],
                changed_fields=clean_fields,
                new_data=payload or None,
                performed_by=user,
                client_ip=get_client_ip(request),
                user_agent=str(request.META.get("HTTP_USER_AGENT", ""))[:255] if request else "",
            )
    except Exception as exc:
        logger.warning("Audit log skipped for %s #%s: %s", table_name, record_id, exc)


class AuditedModelViewSet(viewsets.ModelViewSet):
    """Automatically records real CREATE, UPDATE, and DELETE entries in audit_logs."""

    def perform_create(self, serializer):
        instance = serializer.save()
        record_audit_log(
            request=self.request,
            table_name=instance._meta.db_table,
            record_id=instance.pk or 0,
            action="CREATE",
            changed_fields=list(serializer.validated_data.keys()),
        )

    def perform_update(self, serializer):
        changed = list(serializer.validated_data.keys())
        instance = serializer.save()
        record_audit_log(
            request=self.request,
            table_name=instance._meta.db_table,
            record_id=instance.pk or 0,
            action="UPDATE",
            changed_fields=changed,
        )

    def perform_destroy(self, instance):
        table_name = instance._meta.db_table
        record_id = instance.pk or 0
        instance.delete()
        record_audit_log(
            request=self.request,
            table_name=table_name,
            record_id=record_id,
            action="DELETE",
        )

import hashlib
import hmac
from dataclasses import dataclass
from django.conf import settings


@dataclass(frozen=True)
class VerifiedQrData:
    lrn: str
    qr_token: str
    is_signed: bool


def sign_student_qr(lrn: str, qr_token: str) -> str:
    clean_lrn = str(lrn or "").strip()
    clean_token = str(qr_token or "").strip()
    payload = f"{clean_lrn}:{clean_token}".encode("utf-8")
    sig = hmac.new(settings.SECRET_KEY.encode("utf-8"), payload, hashlib.sha256).hexdigest()[:12].upper()
    return f"ATTENDSURE:{clean_lrn}:{clean_token}:{sig}"


def verify_student_qr(raw_scan: str, *, allow_legacy_unsigned: bool | None = None) -> VerifiedQrData | None:
    if allow_legacy_unsigned is None:
        allow_legacy_unsigned = bool(getattr(settings, "ALLOW_LEGACY_UNSIGNED_QR", False))
    clean = str(raw_scan or "").strip().replace("\r", "").replace("\n", "").replace("\0", "")[:128]
    if not clean:
        return None

    parts = clean.split(":")
    if len(parts) == 4 and parts[0].upper() == "ATTENDSURE":
        _, lrn, token, provided_sig = parts
        expected_sig = sign_student_qr(lrn, token).rsplit(":", 1)[-1]
        if not hmac.compare_digest(provided_sig.upper(), expected_sig):
            return None
        return VerifiedQrData(lrn=lrn.strip(), qr_token=token.strip(), is_signed=True)

    if not allow_legacy_unsigned:
        return None

    token_only = parts[-1].strip()
    return VerifiedQrData(lrn=token_only, qr_token=token_only, is_signed=False)

import math


def calculate_distance_meters(lat1, lon1, lat2, lon2) -> float:
    try:
        u_lat, u_lon = float(lat1), float(lon1)
        c_lat, c_lon = float(lat2), float(lon2)
    except (TypeError, ValueError):
        return float("inf")

    if not all(math.isfinite(v) for v in (u_lat, u_lon, c_lat, c_lon)):
        return float("inf")

    earth_radius = 6371000.0
    phi1, phi2 = math.radians(c_lat), math.radians(u_lat)
    d_phi = math.radians(u_lat - c_lat)
    d_lambda = math.radians(u_lon - c_lon)
    a = (math.sin(d_phi / 2.0) ** 2) + math.cos(phi1) * math.cos(phi2) * (math.sin(d_lambda / 2.0) ** 2)
    a = min(1.0, max(0.0, a))
    return round(earth_radius * (2.0 * math.atan2(math.sqrt(a), math.sqrt(1.0 - a))), 1)


def verify_campus_gps(
    *,
    latitude,
    longitude,
    campus_lat,
    campus_lon,
    radius_meters: int = 150,
    accuracy_meters=None,
    is_mock_location: bool = False,
) -> dict:
    if is_mock_location:
        return {
            "is_valid": False,
            "is_inside": False,
            "distance_meters": None,
            "allowed_radius": radius_meters,
            "reason": "Fake or mock GPS location detected.",
        }

    try:
        u_lat, u_lon = float(latitude), float(longitude)
    except (TypeError, ValueError):
        return {
            "is_valid": False,
            "is_inside": False,
            "distance_meters": None,
            "allowed_radius": radius_meters,
            "reason": "Invalid or missing GPS coordinates.",
        }

    if (
        not (math.isfinite(u_lat) and math.isfinite(u_lon))
        or not (-90.0 <= u_lat <= 90.0)
        or not (-180.0 <= u_lon <= 180.0)
        or (u_lat == 0.0 and u_lon == 0.0)
    ):
        return {
            "is_valid": False,
            "is_inside": False,
            "distance_meters": None,
            "allowed_radius": radius_meters,
            "reason": "GPS coordinates are outside valid earth bounds.",
        }

    distance = calculate_distance_meters(u_lat, u_lon, campus_lat, campus_lon)
    if not math.isfinite(distance):
        return {
            "is_valid": False,
            "is_inside": False,
            "distance_meters": None,
            "allowed_radius": radius_meters,
            "reason": "Campus geofence coordinates are not configured.",
        }

    try:
        parsed_acc = float(accuracy_meters) if accuracy_meters not in (None, "") else 0.0
        if not math.isfinite(parsed_acc):
            parsed_acc = 0.0
    except (TypeError, ValueError):
        parsed_acc = 0.0

    drift_buffer = min(max(parsed_acc, 0.0), 35.0)
    is_inside = distance <= (radius_meters + drift_buffer)

    return {
        "is_valid": True,
        "is_inside": is_inside,
        "distance_meters": distance,
        "allowed_radius": radius_meters,
        "reason": "Within campus boundary." if is_inside else f"Outside campus ({round(distance)}m away).",
    }

from rest_framework import permissions

LEADERSHIP_ROLES = {"ADMIN", "SUPERADMIN", "PRINCIPAL", "DEPT_HEAD"}


def get_user_role(user) -> str:
    if not user or not user.is_authenticated:
        return "ANONYMOUS"
    if user.is_superuser:
        return "ADMIN"
    profile = getattr(user, "profile", None)
    if profile and profile.role:
        return str(profile.role).upper()
    return "ADMIN" if user.is_staff else "FACULTY"


def is_school_leader(user) -> bool:
    return bool(user and user.is_authenticated and (user.is_superuser or get_user_role(user) in LEADERSHIP_ROLES))


class IsLeadershipOrReadOnly(permissions.BasePermission):
    def has_permission(self, request, view):
        if not request.user or not request.user.is_authenticated:
            return False
        if request.method in permissions.SAFE_METHODS:
            return True
        return is_school_leader(request.user)


class IsSystemAdmin(permissions.BasePermission):
    def has_permission(self, request, view):
        return bool(
            request.user
            and request.user.is_authenticated
            and (request.user.is_superuser or get_user_role(request.user) in {"ADMIN", "SUPERADMIN"})
        )

import logging
from rest_framework import status
from rest_framework.response import Response
from rest_framework.views import exception_handler

logger = logging.getLogger(__name__)


class AppError(Exception):
    def __init__(self, error_code: str, message: str, status_code: int = 400):
        self.error_code = error_code
        self.message = message
        self.status_code = status_code
        super().__init__(message)


def custom_exception_handler(exc, context):
    if isinstance(exc, AppError):
        return Response(
            {"ok": False, "error_code": exc.error_code, "message": exc.message},
            status=exc.status_code,
        )

    response = exception_handler(exc, context)
    if response is not None:
        response.data = {
            "ok": False,
            "error_code": "VALIDATION_ERROR",
            "details": response.data,
        }
        return response

    view_name = context.get("view").__class__.__name__ if context.get("view") else "UnknownView"
    logger.exception("Unhandled server error in %s: %s", view_name, exc)
    return Response(
        {
            "ok": False,
            "error_code": "INTERNAL_SERVER_ERROR",
            "message": "An unexpected server error occurred.",
        },
        status=status.HTTP_500_INTERNAL_SERVER_ERROR,
    )

import base64
import hashlib
import io
import os
import uuid
from django.conf import settings
from django.core.files.base import ContentFile
from PIL import Image, UnidentifiedImageError
from rest_framework import serializers

MAX_IMAGE_BYTES = 5 * 1024 * 1024
MAX_IMAGE_DIMENSION = 1024
ALLOWED_IMAGE_EXTS = {"png", "jpg", "jpeg", "webp"}


def save_base64_logo_to_disk(raw_value: str) -> str | None:
    """Converts raw Base64 logo strings into small static files under /media/school_logos/."""
    clean = str(raw_value or "").strip()
    if not clean or clean.lower() in ("none", "null", "undefined"):
        return None

    if clean.startswith(("http://", "https://", "/media/")):
        return clean
    if clean.startswith("school_logos/"):
        return f"/media/{clean}"

    if clean.startswith("data:image") or clean.startswith(("/9j/", "iVBORw0KGgo")):
        try:
            if ";base64," in clean:
                header, b64_part = clean.split(";base64,", 1)
                raw_ext = header.split("/")[-1].split(";")[0].lower()
                ext = "jpg" if raw_ext in ("jpeg", "pjpeg") else raw_ext
            else:
                b64_part = clean
                ext = "jpg" if clean.startswith("/9j/") else "png"

            if ext not in ALLOWED_IMAGE_EXTS:
                ext = "png"

            raw_bytes = base64.b64decode(b64_part, validate=True)
            if len(raw_bytes) > MAX_IMAGE_BYTES:
                return None

            with Image.open(io.BytesIO(raw_bytes)) as img:
                img.verify()

            digest = hashlib.sha256(raw_bytes).hexdigest()[:16]
            target_dir = os.path.join(settings.MEDIA_ROOT, "school_logos")
            os.makedirs(target_dir, exist_ok=True)
            filename = f"logo_{digest}.{ext}"
            full_path = os.path.join(target_dir, filename)
            if not os.path.exists(full_path):
                with open(full_path, "wb") as fh:
                    fh.write(raw_bytes)
            return f"/media/school_logos/{filename}"
        except Exception:
            return None
    return None


def build_media_url(photo_field, request=None) -> str | None:
    if not photo_field:
        return None
    if isinstance(photo_field, str):
        clean = photo_field.strip()
        if not clean or clean.lower() in ("none", "null", "undefined"):
            return None
        if clean.startswith("data:image") or clean.startswith(("/9j/", "iVBORw0KGgo")):
            disk_rel = save_base64_logo_to_disk(clean)
            if not disk_rel:
                return None
            return request.build_absolute_uri(disk_rel) if request else disk_rel
        if clean.startswith(("http://", "https://")):
            return clean
        rel = clean if clean.startswith("/") else f"/{clean}"
        return request.build_absolute_uri(rel) if request else rel

    if hasattr(photo_field, "url"):
        try:
            return request.build_absolute_uri(photo_field.url) if request else photo_field.url
        except ValueError:
            return None
    return None


def validate_and_optimize_image(raw_bytes: bytes, prefix: str = "photo") -> ContentFile:
    """Verifies real image magic bytes via Pillow and compresses oversized uploads."""
    if len(raw_bytes) > MAX_IMAGE_BYTES:
        raise serializers.ValidationError("Image exceeds the 5 MB maximum size.")

    try:
        with Image.open(io.BytesIO(raw_bytes)) as probe:
            probe.verify()

        with Image.open(io.BytesIO(raw_bytes)) as img:
            has_alpha = img.mode in ("RGBA", "LA") or (img.mode == "P" and "transparency" in img.info)
            if max(img.size) > MAX_IMAGE_DIMENSION:
                img.thumbnail((MAX_IMAGE_DIMENSION, MAX_IMAGE_DIMENSION), Image.Resampling.LANCZOS)

            buffer = io.BytesIO()
            if has_alpha:
                img.save(buffer, format="PNG", optimize=True)
                ext = "png"
            else:
                rgb = img.convert("RGB")
                rgb.save(buffer, format="JPEG", quality=82, optimize=True)
                ext = "jpg"
            return ContentFile(buffer.getvalue(), name=f"{prefix}_{uuid.uuid4().hex[:12]}.{ext}")
    except (UnidentifiedImageError, OSError) as exc:
        raise serializers.ValidationError("Invalid or corrupted image file.") from exc


def decode_uploaded_image(data, prefix: str = "upload"):
    if data in (None, "", "null", "undefined"):
        return None
    if hasattr(data, "read"):
        raw_bytes = data.read()
        data.seek(0)
        return validate_and_optimize_image(raw_bytes, prefix=prefix)
    if isinstance(data, str):
        clean = data.strip()
        if clean.startswith("data:image") and ";base64," in clean:
            _, b64_part = clean.split(";base64,", 1)
            try:
                raw_bytes = base64.b64decode(b64_part, validate=True)
            except Exception as exc:
                raise serializers.ValidationError("Invalid Base64 image data.") from exc
            return validate_and_optimize_image(raw_bytes, prefix=prefix)
    return serializers.SkipField()


class FlexibleImageField(serializers.Field):
    def to_representation(self, value):
        return build_media_url(value, self.context.get("request"))

    def to_internal_value(self, data):
        result = decode_uploaded_image(data, prefix="photo")
        if isinstance(result, serializers.SkipField):
            raise result
        return result


class SchoolLogoField(serializers.Field):
    def to_representation(self, value):
        return build_media_url(value, self.context.get("request"))

    def to_internal_value(self, data):
        file_obj = decode_uploaded_image(data, prefix="logo")
        if isinstance(file_obj, serializers.SkipField):
            raise file_obj
        if file_obj is None:
            return None
        raw_bytes = file_obj.read()
        ext = getattr(file_obj, "name", "logo.png").rsplit(".", 1)[-1].lower()
        digest = hashlib.sha256(raw_bytes).hexdigest()[:16]
        target_dir = os.path.join(settings.MEDIA_ROOT, "school_logos")
        os.makedirs(target_dir, exist_ok=True)
        filename = f"logo_{digest}.{ext}"
        with open(os.path.join(target_dir, filename), "wb") as fh:
            fh.write(raw_bytes)
        return f"/media/school_logos/{filename}"

        db_table = "user_profiles"

    def is_locked(self) -> bool:
        return bool(self.locked_until and timezone.now() < self.locked_until)

    def record_failed_attempt(self):
        now = timezone.now()
        if self.locked_until and now >= self.locked_until:
            UserProfile.objects.filter(pk=self.pk).update(failed_attempts=1, locked_until=None)
            self.failed_attempts = 1
            self.locked_until = None
            return
        UserProfile.objects.filter(pk=self.pk).update(failed_attempts=F("failed_attempts") + 1)
        self.refresh_from_db(fields=["failed_attempts"])
        if self.failed_attempts >= 5:
            self.locked_until = now + timezone.timedelta(minutes=15)
            self.save(update_fields=["locked_until"])

    def reset_failed_attempts(self):
        if self.failed_attempts > 0 or self.locked_until is not None:
            UserProfile.objects.filter(pk=self.pk).update(failed_attempts=0, locked_until=None)

    def save(self, *args, **kwargs):
        if self.phone_number:
            self.phone_number = normalize_ph_phone(self.phone_number)
        if self.role == "PARENT":
            self.is_teaching = False
        super().save(*args, **kwargs)

from django.contrib.auth.models import User
from django.core.cache import cache

def resolve_teacher_for_user(user: User | None) -> TeacherProfile | None:
    if not user or not getattr(user, "is_authenticated", False):
        return None

    if get_user_role(user) == "PARENT":
        return None

    cache_key = f"accounts:teacher_for_user:{user.id}"
    cached_id = cache.get(cache_key)
    if cached_id is not None:
        if cached_id == 0:
            return None
        found = TeacherProfile.objects.select_related("account").filter(id=cached_id, is_active=True).first()
        if found:
            return found

    profile = UserProfile.objects.select_related("teacher").filter(user=user).first()
    if profile and getattr(profile, "teacher", None):
        teacher = profile.teacher
        if teacher and teacher.is_active:
            cache.set(cache_key, teacher.id, timeout=60)
            return teacher

    by_account = TeacherProfile.objects.select_related("account").filter(account__user=user, is_active=True).first()
    if by_account:
        cache.set(cache_key, by_account.id, timeout=60)
        return by_account

    lookup_q = Q()
    clean_username = str(getattr(user, "username", "") or "").strip()
    clean_email = str(getattr(user, "email", "") or "").strip()
    if clean_username:
        lookup_q |= Q(employee_id__iexact=clean_username)
    if clean_email:
        lookup_q |= Q(email__iexact=clean_email)

    by_username = (
        TeacherProfile.objects.select_related("account")
        .filter(lookup_q, is_active=True)
        .first()
        if lookup_q
        else None
    )
    if by_username:
        cache.set(cache_key, by_username.id, timeout=60)
        return by_username

    cache.set(cache_key, 0, timeout=30)
    return None


def get_teacher_section_ids(teacher: TeacherProfile | None) -> set[int]:
    if not teacher:
        return set()

    cache_key = f"accounts:teacher_sections:{teacher.id}"
    cached = cache.get(cache_key)
    if cached is not None:
        return set(cached)

    advisory_ids = set(
        Section.objects.filter(adviser=teacher, is_active=True).values_list("id", flat=True)
    )
    period_sec_ids = set(
        ClassPeriod.objects.filter(teacher=teacher, is_active=True, section__is_active=True)
        .exclude(section_id__isnull=True)
        .values_list("section_id", flat=True)
    )
    combined = advisory_ids | period_sec_ids
    cache.set(cache_key, list(combined), timeout=30)
    return combined


def get_teacher_student_ids(teacher: TeacherProfile | None) -> set[int]:
    if not teacher:
        return set()

    section_ids = get_teacher_section_ids(teacher)
    if not section_ids:
        return set()

    return set(
        Enrollment.objects.filter(
            section_id__in=section_ids,
            status__in=["ENROLLED", "TRANSFERRED_IN"],
            student__is_active=True,
        ).values_list("student_id", flat=True)
    )


def build_user_session_payload(user: User, request=None) -> dict:
    """Delegates to the canonical session payload builder in accounts.services."""

    def create(self, validated_data):

        is_admin_role = raw_role in {"ADMIN", "SUPERADMIN", "FACULTY_ADMIN"}
        is_teaching = False if role == "PARENT" else bool(validated_data.pop("is_teaching", True))
        emp_id = str(validated_data.pop("employee_id", "") or validated_data["username"]).strip().upper()[:50]

        existing_teacher = None
        if role != "PARENT":
            existing_teacher = TeacherProfile.objects.select_related("account").filter(employee_id__iexact=emp_id).first()
            if existing_teacher and getattr(existing_teacher, "account", None) is not None:
                raise serializers.ValidationError(
                    {"employee_id": "This Employee ID is already linked to another user account."}
                )

        user = User(**validated_data, is_staff=is_admin_role, is_superuser=is_admin_role)
        if password:
            user.set_password(password)
        else:
            user.set_unusable_password()
        user.save()

        profile, _ = UserProfile.objects.update_or_create(
            user=user,
            defaults={
                "role": role,
                "phone_number": phone,
                "is_teaching": is_teaching,
                "is_profile_completed": bool(user.first_name and user.last_name),
                **({"avatar": photo} if photo else {}),

        )

        if role != "PARENT":

                teacher.first_name = (user.first_name or teacher.first_name or user.username)[:100]
                if middle_name:

                    teacher.position = pos

            else:

                    employee_id=emp_id,

                    middle_name=middle_name,
                    last_name=(user.last_name or "Staff")[:100],
                    suffix=suffix,
                    email=(user.email or "")[:150],
                    contact_number=phone,
                    department=dept,
                    position=pos,
                    can_teach=is_teaching,
                    is_active=user.is_active,
                    qr_token=secrets.token_urlsafe(24),
                )
            profile.teacher = teacher
            profile.save(update_fields=["teacher"])
        return user

    @transaction.atomic
    def update(self, instance, validated_data):
        request = self.context.get("request")
        is_admin = bool(
            request and request.user.is_authenticated
            and (request.user.is_superuser or get_user_role(request.user) in {"ADMIN", "SUPERADMIN"})
        )

        password = validated_data.pop("password", "")
        raw_role = validated_data.pop("role", None)
        is_teaching = validated_data.pop("is_teaching", None)
        middle_name = validated_data.pop("middle_name", None)
        suffix = validated_data.pop("suffix", None)
        phone = validated_data.pop("contact_number", None)
        dept = validated_data.pop("department", None)
        pos = validated_data.pop("position", None)
        validated_data.pop("employee_id", None)
        photo = validated_data.pop("photo", None)

        if not is_admin:
            validated_data.pop("is_active", None)
            raw_role = None

        for attr, val in validated_data.items():
            setattr(instance, attr, val)
        if password:
            instance.set_password(password)
        instance.save()

        if password:
            Token.objects.filter(user=instance).delete()

        profile, _ = UserProfile.objects.get_or_create(user=instance)
        if raw_role and is_admin:
            clean_role = str(raw_role).upper()
            profile.role = "ADMIN" if clean_role in {"FACULTY_ADMIN", "SUPERADMIN"} else clean_role
            is_admin_role = clean_role in {"ADMIN", "SUPERADMIN", "FACULTY_ADMIN"}
            instance.is_staff = is_admin_role
            instance.is_superuser = is_admin_role
            instance.save(update_fields=["is_staff", "is_superuser"])

        if is_teaching is not None and is_admin:
            profile.is_teaching = False if profile.role == "PARENT" else bool(is_teaching)
        if phone is not None:
            profile.phone_number = normalize_ph_phone(phone)
        if photo:
            profile.avatar = photo
        profile.is_profile_completed = bool(instance.first_name and instance.last_name)
        profile.save()

        teacher_id = None
        if profile.teacher:
            t = profile.teacher
            teacher_id = t.id
            t.first_name = (instance.first_name or t.first_name)[:100]
            if middle_name is not None:
                t.middle_name = str(middle_name).strip()[:100]
            t.last_name = (instance.last_name or t.last_name)[:100]
            if suffix is not None:
                t.suffix = str(suffix).strip()[:20]

            if phone is not None:
                t.contact_number = profile.phone_number
            if dept is not None:
                t.department = str(dept)[:100]
            if pos is not None:
                t.position = str(pos)[:100]
            t.can_teach = profile.is_teaching
            t.is_active = instance.is_active
            t.save()

        invalidate_user_and_teacher_caches(user_id=instance.id, teacher_id=teacher_id)
        return instance

"""
ATTENDSURE - Gunicorn Production Configuration
Formula for workers: (2 x CPU cores) + 1
Example: 4 CPU cores = 9 workers
"""

import multiprocessing

# Bind to internal port (Nginx will talk to this)
bind = "127.0.0.1:8000"

# Number of worker processes handling requests
workers = (multiprocessing.cpu_count() * 2) + 1

# Worker type
worker_class = "sync"

# Maximum time a worker can take to answer before restarting (prevents frozen requests)
timeout = 30

# Keep-alive connections for fast mobile requests
keepalive = 5

# Restart workers periodically to prevent memory leaks from photo processing
max_requests = 1000
max_requests_jitter = 100

# Logging
accesslog = "-"
errorlog = "-"
loglevel = "info"

    }

    server {
        listen 80;
        server_name api.attendsure.ph 10.16.115.49;

        client_max_body_size 20M;

        location / {
            proxy_pass http://attendsure_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Zero-downtime timeouts
            proxy_connect_timeout 5s;
            proxy_read_timeout 30s;
        }
    }
}
