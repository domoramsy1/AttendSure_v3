r"""
AttendSure - Terminate Stuck Idle PostgreSQL Connections
Run: python database\reset_connections.py
r"""

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from _bootstrap import setup_django

setup_django()

from django.db import connection


def main():
    if connection.vendor != "postgresql":
        print(f"Skipped: active engine is {connection.vendor}.")
        return
    with connection.cursor() as cursor:
        cursor.execute("""
            SELECT pg_terminate_backend(pid)
            FROM pg_stat_activity
            WHERE datname = current_database()
              AND pid <> pg_backend_pid()
              AND state = 'idle';
        """)
        terminated = cursor.rowcount
    print(f"[OK] Terminated {terminated} idle PostgreSQL connection(s).")


if __name__ == "__main__":
    main()

PALETTES = [
    ("#1e3a8a", "#3b82f6"),
    ("#065f46", "#10b981"),
    ("#0f172a", "#64748b"),
    ("#312e81", "#6366f1"),
]


def draw_avatar(target_path: Path, initials: str, seed_num: int, size: int = 256) -> None:
    target_path.parent.mkdir(parents=True, exist_ok=True)
    bg, border = PALETTES[seed_num % len(PALETTES)]
    img = Image.new("RGB", (size, size), bg)
    draw = ImageDraw.Draw(img)
    draw.ellipse([(8, 8), (size - 8, size - 8)], fill=bg, outline=border, width=6)

    label = (initials or "AS")[:2].upper()
    font = ImageFont.load_default()
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = max(1, bbox[2] - bbox[0]), max(1, bbox[3] - bbox[1])

    glyph = Image.new("RGBA", (tw + 4, th + 4), (0, 0, 0, 0))
    ImageDraw.Draw(glyph).text((2, 2), label, fill=(255, 255, 255, 255), font=font)
    scaled = glyph.resize((int(size * 0.42), int(size * 0.42)), Image.Resampling.NEAREST)
    img.paste(scaled, ((size - scaled.width) // 2, (size - scaled.height) // 2), scaled)
    img.save(target_path, "JPEG", quality=88)


def main():
    media_root = Path(settings.MEDIA_ROOT)
    st_fixed = tc_fixed = us_fixed = 0

    for st in Student.objects.only("id", "lrn", "first_name", "last_name", "photo"):
        rel = st.photo.name if (st.photo and st.photo.name) else f"students/std_{st.lrn or st.id}.jpg"
        full = media_root / rel
        if not full.is_file():
            draw_avatar(full, f"{st.first_name[:1]}{st.last_name[:1]}", st.id)
            if not st.photo or st.photo.name != rel:
                Student.objects.filter(pk=st.id).update(photo=rel)
            st_fixed += 1

    for tc in TeacherProfile.objects.only("id", "employee_id", "first_name", "last_name", "photo"):
        rel = tc.photo.name if (tc.photo and tc.photo.name) else f"faculty/fac_{tc.employee_id or tc.id}.jpg"
        full = media_root / rel
        if not full.is_file():
            draw_avatar(full, f"{tc.first_name[:1]}{tc.last_name[:1]}", tc.id)
            if not tc.photo or tc.photo.name != rel:
                TeacherProfile.objects.filter(pk=tc.id).update(photo=rel)
            tc_fixed += 1

    for prof in UserProfile.objects.select_related("user", "teacher"):
        if prof.teacher and prof.teacher.photo:
            continue
        rel = prof.avatar.name if (prof.avatar and prof.avatar.name) else f"avatars/user_{prof.user_id}.jpg"
        full = media_root / rel
        if not full.is_file():
            u = prof.user
            initials = f"{u.first_name[:1]}{u.last_name[:1]}".strip() or u.username[:2]
            draw_avatar(full, initials, prof.user_id)
            if not prof.avatar or prof.avatar.name != rel:
                UserProfile.objects.filter(pk=prof.id).update(avatar=rel)
            us_fixed += 1

    print(f"[OK] Avatars verified -> Students: {st_fixed}, Teachers: {tc_fixed}, Users: {us_fixed} created.")


if __name__ == "__main__":

class ApiSuiteVerifier:
    def __init__(self):
        self.client = APIClient()
        self.results = []

        self.tag = uuid.uuid4().hex[:6].upper()

    def _err_text(self, res) -> str:
        data = getattr(res, "data", None)
        if data is not None:
            return str(data)[:300]
        return getattr(res, "content", b"").decode("utf-8", errors="replace")[:300]

    def record(self, cat: str, name: str, ok: bool, detail: str, ms: float, err: str = ""):
        self.results.append({"passed": ok})
        badge = "PASS" if ok else "FAIL"
        print(f"[{badge}] {cat:<14} | {name:<36} | {ms:>6.1f} ms | {detail}")
        if not ok:
            self.errors.append(f"[{cat}] {name} -> {detail} | {err}")

    def call(self, fn, *args, **kwargs):
        t0 = time.perf_counter()
        res = fn(*args, **kwargs)
        return res, (time.perf_counter() - t0) * 1000.0

    def authenticate_admin(self) -> bool:
        admin = User.objects.filter(is_superuser=True, is_active=True).first()
        if not admin:
            self.record("AUTH", "Superuser Lookup", False, "No active superuser in DB", 0.0)
            return False

        self.client.credentials(HTTP_AUTHORIZATION=f"Token {token.key}", HTTP_HOST="localhost")
        return True

    def crud_cycle(self, label: str, url: str, create_data: dict, patch_data: dict, check_key: str, query: str = ""):

def _safe_rel(path: Path, base: Path) -> str:

def is_ignored_file(path: Path) -> bool:
    if path.name in IGNORED_FILES or path.name.endswith("_codes_export.txt"):
        return True
    if path.suffix.lower() in BINARY_EXTENSIONS:
        return True
    try:
        if path.stat().st_size > MAX_FILE_BYTES:
            return True
        with path.open("rb") as fh:
            return b"\x00" in fh.read(4096)
    except OSError:
        return True


def get_visible_entries(directory: Path, exclude_env: bool = False):
    try:
        raw_entries = list(directory.iterdir())
    except OSError:
        return [], []
    dirs = sorted([d for d in raw_entries if d.is_dir() and d.name not in IGNORED_DIRS], key=lambda p: p.name.lower())
    files = sorted(
        [f for f in raw_entries if f.is_file() and not is_ignored_file(f) and not (exclude_env and f.name == ".env")],
        key=lambda p: p.name.lower(),
    )
    return dirs, files


def collect_tree_and_files(target: Path, exclude_env: bool = False):
    tree_lines, collected_files, folder_stats = [], [], {}
    if target.is_file():
        tree_lines.append(f"{PROJECT_ROOT.name}/")
        tree_lines.append(f"+-- {_safe_rel(target, PROJECT_ROOT)}")
        return tree_lines, [target], {str(target.parent): 1}, 1

    prefix = ""
    if target == PROJECT_ROOT:
        tree_lines.append(f"{PROJECT_ROOT.name}/")
    elif PROJECT_ROOT in target.parents:
        tree_lines.append(f"{PROJECT_ROOT.name}/")
        for part in target.relative_to(PROJECT_ROOT).parts:
            tree_lines.append(f"{prefix}+-- {part}/")
            prefix += "    "
    else:
        tree_lines.append(f"{target.name}/")

    folder_count = 0

    def _walk(curr_dir: Path, curr_prefix: str):
        nonlocal folder_count
        dirs, files = get_visible_entries(curr_dir, exclude_env=exclude_env)
        folder_stats[str(curr_dir)] = len(files)
        entries = dirs + files
        for idx, entry in enumerate(entries):
            is_last = idx == len(entries) - 1
            conn = "+-- " if is_last else "|-- "
            if entry.is_dir():
                folder_count += 1
                tree_lines.append(f"{curr_prefix}{conn}{entry.name}/")
                _walk(entry, curr_prefix + ("    " if is_last else "|   "))
            else:
                tree_lines.append(f"{curr_prefix}{conn}{entry.name}")
                collected_files.append(entry)

    _walk(target, prefix)
    return tree_lines, collected_files, folder_stats, folder_count


def format_size(num_bytes: int) -> str:
    if num_bytes >= 1024 * 1024:
        return f"{num_bytes / (1024 * 1024):.2f} MB"
    if num_bytes >= 1024:
        return f"{num_bytes / 1024:.1f} KB"
    return f"{num_bytes} B"

        total_lines += lc
        records.append((fpath, text, lc, len(raw)))

    total_files = len(records)
    pad = max(3, len(str(total_files)))
    sep = "=" * 80

        sep, "", sep, "SECTION 1: VISUAL DIRECTORY STRUCTURE", sep, *tree_lines,
        "", sep, "SECTION 2A: FOLDER SUMMARY", sep,
    ]
    for f_dir, cnt in folder_stats.items():
        rel_f = _safe_rel(Path(f_dir), PROJECT_ROOT)

    summary.extend(["", sep, f"SECTION 2B: NUMBERED COMPLETE FILE PATH INDEX (1 to {total_files})", sep])
    for idx, (fp, _, lc, sb) in enumerate(records, start=1):

    summary_text = "\n".join(summary)
    print("\n" + summary_text)

    codes = ["", sep, f"SECTION 3: COMPLETE FILE CODES (1 to {total_files})", sep]
    for idx, (fp, content, lc, sb) in enumerate(records, start=1):
