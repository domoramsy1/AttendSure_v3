class ApiSecurityMiddleware:
    """
    Applies OWASP security headers and prevents 40 KB Django debug HTML pages
    from leaking internal URL patterns when an /api/ path returns 404.
    """

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        response = self.get_response(request)

        if request.path.startswith("/api/") and response.status_code == 404:
            content_type = response.get("Content-Type", "")
            if "text/html" in content_type:
                response = JsonResponse(
                    {
                        "ok": False,
                        "error_code": "ENDPOINT_NOT_FOUND",
                        "message": f"API route not found: {request.path}",
                    },
                    status=404,
                )

        response["X-Content-Type-Options"] = "nosniff"
        response["X-Frame-Options"] = "DENY"
        response["Referrer-Policy"] = "same-origin"
        response["Permissions-Policy"] = "camera=(self), geolocation=(self), microphone=()"

        if request.path.startswith(("/api/auth/", "/api/v1/accounts/login", "/api/me")):
            response["Cache-Control"] = "no-store, no-cache, must-revalidate, private"

        return response


@api_view(["GET"])
@permission_classes([permissions.AllowAny])
def database_health_view(request):
    """Checks live PostgreSQL connectivity without exposing database names or hosts."""
    started = time.perf_counter()
    with connection.cursor() as cursor:
        cursor.execute("SELECT 1;")
        cursor.fetchone()
    latency_ms = max(1, round((time.perf_counter() - started) * 1000))
    return Response(
        {
            "ok": True,
            "status": "CONNECTED",
            "database": "CONNECTED",
            "latency_ms": latency_ms,
        },
        status=status.HTTP_200_OK,
    )


@api_view(["POST"])
@permission_classes([permissions.IsAuthenticated, IsSystemAdmin])
def clear_system_cache_view(request):
    """Clears server cache for system administrators only and logs the action."""
    cache.clear()
    record_audit_log(
        request=request,
        table_name="system_cache",
        record_id=0,
        action="FLUSH",
    )
    return Response(
        {"ok": True, "success": True, "message": "System cache cleared."},
        status=status.HTTP_200_OK,
    )

SCHOOL_META_CACHE_KEY = "school:metadata:raw"


def format_section_label(grade_name: str, section_name: str) -> str:
    g, s = str(grade_name or "").strip(), str(section_name or "").strip()
    if not g:
        return s
    if not s or g.lower().endswith(s.lower()):
        return g
    return f"{g} - {s}"


def migrate_inline_logos_to_files(school: SchoolSettings) -> SchoolSettings:
    """Replaces raw Base64 logo strings in PostgreSQL with lightweight /media/ paths."""
    updates = {}
    for field_name in ("school_logo", "left_logo", "right_logo"):
        val = getattr(school, field_name, None)
        if val and isinstance(val, str) and (
            val.startswith("data:image") or val.startswith(("/9j/", "iVBORw0KGgo"))
        ):
            disk_path = save_base64_logo_to_disk(val)
            if disk_path:
                updates[field_name] = disk_path
                setattr(school, field_name, disk_path)

    if updates:
        SchoolSettings.objects.filter(pk=school.pk).update(**updates)
        cache.delete(SCHOOL_META_CACHE_KEY)
    return school


def get_school_metadata(request=None, school: SchoolSettings | None = None) -> dict:
    raw = cache.get(SCHOOL_META_CACHE_KEY)
    if raw is None:
        school = school or SchoolSettings.objects.select_related("principal_teacher").first()
        if not school:
            return {"school_id": "", "school_name": "", "is_configured": False}

        school = migrate_inline_logos_to_files(school)
        head_name = school.principal_name or (
            school.principal_teacher.full_name if school.principal_teacher else ""
        )
        raw = {
            "school_id": school.school_id,
            "school_name": school.school_name,
            "region": school.region,
            "division": school.division,
            "district": school.district,
            "academic_year": school.academic_year,
            "school_address": school.address,
            "school_contact": school.contact_number,
            "school_email": school.email,
            "website": school.website,
            "school_head": head_name.upper(),
            "principal_title": school.principal_title,
            "superintendent_name": school.superintendent_name,
            "left_logo_path": school.left_logo,
            "right_logo_path": school.right_logo,
            "school_logo_path": school.school_logo,
            "is_configured": bool(school.school_name and school.school_id),
        }
        cache.set(SCHOOL_META_CACHE_KEY, raw, timeout=300)

    return {
        "school_id": raw["school_id"],
        "school_name": raw["school_name"],
        "region": raw["region"],
        "division": raw["division"],
        "district": raw["district"],
        "academic_year": raw["academic_year"],
        "school_address": raw["school_address"],
        "school_contact": raw["school_contact"],
        "school_email": raw["school_email"],
        "website": raw["website"],
        "school_head": raw["school_head"],
        "principal_title": raw["principal_title"],
        "superintendent_name": raw["superintendent_name"],
        "left_logo": build_media_url(raw["left_logo_path"], request),
        "right_logo": build_media_url(raw["right_logo_path"], request),
        "school_logo": build_media_url(raw["school_logo_path"], request),
        "is_configured": raw["is_configured"],
    }


def get_sections_for_user(user, *, include_inactive: bool = False):
    qs = Section.objects.all()
    if not (include_inactive and is_school_leader(user)):
        qs = qs.filter(is_active=True)
    qs = (
        qs.select_related("grade_level", "school_year", "adviser", "room")
        .annotate(
            enrolled_count=Count(
                "enrollments",
                filter=Q(
                    enrollments__status__in=["ENROLLED", "TRANSFERRED_IN"],
                    enrollments__student__is_active=True,
                ),
                distinct=True,
            )
        )
        .order_by("grade_level__level_order", "name")
    )
    if not is_school_leader(user):
        teacher = resolve_teacher_for_user(user)
        qs = qs.filter(id__in=get_teacher_section_ids(teacher)) if teacher else qs.none()
    return qs


def get_class_periods_for_user(user, section_id: int | None = None):
    qs = ClassPeriod.objects.select_related(
        "section", "section__grade_level", "subject", "teacher", "room"
    ).order_by("start_time")
    if section_id:
        qs = qs.filter(section_id=section_id)
    if not is_school_leader(user):
        teacher = resolve_teacher_for_user(user)
        if teacher:
            qs = qs.filter(
                Q(teacher=teacher) | Q(section_id__in=get_teacher_section_ids(teacher))
            ).distinct()
        else:
            qs = qs.none()
    return qs

class SchoolHolidaySerializer(serializers.ModelSerializer):
    date = serializers.DateField(source="holiday_date")
    reason = serializers.CharField(source="title", required=False, write_only=True)
    type = serializers.CharField(required=False, write_only=True, default="HOLIDAY")

    class Meta:
        model = SchoolHoliday
        fields = ["id", "date", "title", "reason", "type", "created_at"]
        extra_kwargs = {"title": {"required": False}}

    def to_internal_value(self, data):
        mutable = {k: data.get(k) for k in data.keys()} if hasattr(data, "keys") else dict(data)
        if not mutable.get("title") and mutable.get("reason"):
            mutable["title"] = mutable["reason"]
        exc_type = str(mutable.get("type") or "").strip().upper()
        raw_title = str(mutable.get("title") or "").strip()
        if exc_type in ("SUSPENSION", "HOLIDAY", "EVENT") and raw_title and not raw_title.startswith("["):
            mutable["title"] = f"[{exc_type}] {raw_title}"[:150]
        elif raw_title:
            mutable["title"] = raw_title[:150]
        return super().to_internal_value(mutable)

    def validate(self, attrs):
        if not str(attrs.get("title") or "").strip():
            raise serializers.ValidationError({"title": "A holiday or calendar override title/reason is required."})
        attrs.pop("type", None)
        return attrs

    def to_representation(self, instance):
        data = super().to_representation(instance)
        raw_title = str(instance.title or "")
        exc_type = "HOLIDAY"
        clean_reason = raw_title
        for prefix in ("SUSPENSION", "HOLIDAY", "EVENT"):
            tag = f"[{prefix}] "
            if raw_title.upper().startswith(tag):
                exc_type = prefix
                clean_reason = raw_title[len(tag):]
                break
        data["type"] = exc_type
        data["reason"] = clean_reason
        data["affects_all_grades"] = True
        return data

    )
    principal_photo_url = serializers.SerializerMethodField()
    principal_employee_id = serializers.SerializerMethodField()
    principal_department = serializers.SerializerMethodField()
    principal_email = serializers.SerializerMethodField()
    principal_contact_number = serializers.SerializerMethodField()

    class Meta:
        model = SchoolSettings
        fields = [
            "id", "school_id", "school_name", "region", "division", "district",
            "academic_year", "address", "contact_number", "email", "website",
            "principal_teacher", "principal_faculty", "principal_name",
            "principal_title", "superintendent_name",
            "principal_photo_url", "principal_employee_id", "principal_department",
            "principal_email", "principal_contact_number",
            "school_logo", "left_logo", "right_logo",
            "latitude", "longitude", "geofence_radius_meters", "updated_at",
        ]
        extra_kwargs = {
            "school_id": {"required": False, "allow_blank": True},
            "school_name": {"required": False, "allow_blank": True},
            "principal_teacher": {"required": False, "allow_null": True},
        }

    def to_internal_value(self, data):
        mutable = {}
        if hasattr(data, "keys"):
            for key in data.keys():
                mutable[key] = data.get(key)
        else:
            mutable = dict(data)

        alias_map = {
            "school_address": "address",
            "school_contact": "contact_number",
            "school_email": "email",
            "school_head": "principal_name",
            "radius_meters": "geofence_radius_meters",
            "geofence_radius": "geofence_radius_meters",
            "emblem": "school_logo",
            "main_logo": "school_logo",
            "principal_faculty": "principal_teacher",
        }
        for src_key, dest_key in alias_map.items():
            if src_key in mutable and dest_key not in mutable:
                mutable[dest_key] = mutable[src_key]
        mutable.pop("principal_faculty", None)

        if "principal_teacher" in mutable and str(mutable["principal_teacher"]).strip().lower() in ("", "null", "none", "undefined"):
            mutable["principal_teacher"] = None

        if "website" in mutable:
            raw_web = str(mutable.get("website") or "").strip()
            if raw_web and not raw_web.startswith(("http://", "https://")):
                mutable["website"] = f"https://{raw_web}"
            else:
                mutable["website"] = raw_web

        for coord_key in ("latitude", "longitude"):
            if coord_key in mutable:
                raw_coord = mutable[coord_key]
                if raw_coord in (None, "", "null", "undefined"):
                    mutable.pop(coord_key, None)
                else:
                    try:
                        parsed = float(raw_coord)
                        if not math.isfinite(parsed):
                            raise ValueError("non-finite")
                        mutable[coord_key] = Decimal(f"{parsed:.7f}")
                    except (TypeError, ValueError, InvalidOperation):
                        raise serializers.ValidationError({coord_key: f"Invalid {coord_key} value."})

        if "geofence_radius_meters" in mutable:
            raw_rad = mutable["geofence_radius_meters"]
            if raw_rad in (None, "", "null", "undefined"):
                mutable.pop("geofence_radius_meters", None)
            else:
                try:
                    parsed_rad = float(raw_rad)
                    if not math.isfinite(parsed_rad):
                        raise ValueError("non-finite")
                    mutable["geofence_radius_meters"] = max(20, min(5000, int(parsed_rad)))
                except (TypeError, ValueError, OverflowError):
                    raise serializers.ValidationError({"geofence_radius_meters": "Invalid radius in meters."})

        return super().to_internal_value(mutable)

    def update(self, instance, validated_data):
        instance = super().update(instance, validated_data)
        if instance.principal_teacher:
            if not instance.principal_name:
                instance.principal_name = instance.principal_teacher.full_name
            if not instance.principal_title and instance.principal_teacher.position:
                instance.principal_title = instance.principal_teacher.position
            instance.save(update_fields=["principal_name", "principal_title", "updated_at"])
        return instance

    def get_principal_photo_url(self, obj):
        if obj.principal_teacher and obj.principal_teacher.photo:
            return build_media_url(obj.principal_teacher.photo, self.context.get("request"))

    def get_principal_employee_id(self, obj) -> str:

    def get_principal_department(self, obj) -> str:

    def get_principal_email(self, obj) -> str:
        return obj.principal_teacher.email if obj.principal_teacher else ""

    def get_principal_contact_number(self, obj) -> str:

    display_label = serializers.SerializerMethodField()
    student_count = serializers.IntegerField(source="enrolled_count", read_only=True, default=0)

    class Meta:
        model = Section
        fields = [
            "id", "name", "code", "grade_level", "grade_level_name",
            "school_year", "school_year_code", "adviser", "adviser_name",
            "room", "room_number", "capacity", "track_strand",
            "display_label", "student_count", "is_active",
        ]
        extra_kwargs = {"school_year": {"required": False}}

    def get_display_label(self, obj) -> str:
        return format_section_label(obj.grade_level.name if obj.grade_level else "", obj.name)

    def validate(self, attrs):
        sy = attrs.get("school_year") or (self.instance.school_year if self.instance else None)
        if not sy:
            sy = SchoolYear.objects.filter(is_active=True).first() or SchoolYear.objects.first()
            if not sy:
                raise serializers.ValidationError({"school_year": "Create an active School Year before adding sections."})
            attrs["school_year"] = sy
        grade_level = attrs.get("grade_level") or (self.instance.grade_level if self.instance else None)
        name = attrs.get("name") if "name" in attrs else (self.instance.name if self.instance else None)
        if sy and grade_level and name:
            dup_qs = Section.objects.filter(school_year=sy, grade_level=grade_level, name=name)
            if self.instance:
                dup_qs = dup_qs.exclude(pk=self.instance.pk)
            if dup_qs.exists():
                raise serializers.ValidationError(
                    {"name": "A section with this name already exists for this grade level and school year."}
                )
        return attrs

    def create(self, validated_data):
        section = super().create(validated_data)
        sync_uniform_subjects_for_section(section)
        return section

from datetime import time
from .models import ClassPeriod, ClassPeriodDay, Section, Subject


def sync_uniform_subjects_for_section(section: Section) -> None:
    if not section or not section.grade_level_id:
        return
    subjects = list(Subject.objects.filter(grade_level_id=section.grade_level_id).order_by("id"))
    existing_ids = set(ClassPeriod.objects.filter(section=section).values_list("subject_id", flat=True))
    to_create = []
    for idx, subj in enumerate(subjects):
        if subj.id in existing_ids:
            continue
        start_hr = min(7 + idx, 16)
        to_create.append(
            ClassPeriod(
                section=section,
                subject=subj,
                start_time=time(start_hr, 30),
                end_time=time(min(start_hr + 1, 17), 30),
                days_of_week="MON-FRI",
                room=section.room,
                room_number=section.room_number or "",
                is_active=True,
            )
        )
    if to_create:
        created_periods = ClassPeriod.objects.bulk_create(to_create, ignore_conflicts=True)
        for period in created_periods:
            if period.pk:
                sync_period_days(period, period.days_of_week)


def sync_period_days(period: ClassPeriod, days_text: str) -> None:
    presets = {"MON-FRI": [1, 2, 3, 4, 5], "MWF": [1, 3, 5], "TTH": [2, 4], "SAT": [6]}
    lookup = {"MON": 1, "TUE": 2, "WED": 3, "THU": 4, "FRI": 5, "SAT": 6, "SUN": 7}
    clean = str(days_text or "MON-FRI").upper().strip()
    raw_nums = presets.get(clean) or [lookup[p.strip()] for p in clean.split(",") if p.strip() in lookup] or [1, 2, 3, 4, 5]
    day_nums = list(dict.fromkeys(raw_nums))

    ClassPeriodDay.objects.filter(class_period=period).exclude(day_of_week__in=day_nums).delete()
    existing = set(ClassPeriodDay.objects.filter(class_period=period).values_list("day_of_week", flat=True))
    ClassPeriodDay.objects.bulk_create(
        [ClassPeriodDay(class_period=period, day_of_week=d) for d in day_nums if d not in existing],
        ignore_conflicts=True,
    )

class SchoolSettingsView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsLeadershipOrReadOnly]

    def get(self, request):
        school = SchoolSettings.objects.select_related("principal_teacher").first()
        if not school:
            return Response(get_school_metadata(request))
        school = migrate_inline_logos_to_files(school)
        data = dict(SchoolSettingsSerializer(school, context={"request": request}).data)
        data.update(get_school_metadata(request, school=school))
        return Response(data)

    @transaction.atomic
    def put(self, request):
        school = SchoolSettings.objects.first()
        if school:
            serializer = SchoolSettingsSerializer(
                school, data=request.data, partial=True, context={"request": request}
            )
        else:
            payload = (
                {k: request.data.get(k) for k in request.data.keys()}
                if hasattr(request.data, "keys")
                else dict(request.data)
            )
            payload.setdefault("school_id", "000000")
            payload.setdefault("school_name", "AttendSure School")
            serializer = SchoolSettingsSerializer(data=payload, context={"request": request})

        serializer.is_valid(raise_exception=True)
        saved = migrate_inline_logos_to_files(serializer.save())
        cache.delete_many([SCHOOL_META_CACHE_KEY, "reports:system_status"])

        radius_val = max(20, int(saved.geofence_radius_meters or 150))
        boundary = (
            CampusBoundary.objects.filter(is_active=True).first()
            or CampusBoundary.objects.first()
        )
        if boundary:
            boundary.name = (saved.school_name or boundary.name or "Campus Boundary")[:100]
            boundary.latitude = saved.latitude
            boundary.longitude = saved.longitude
            boundary.radius_meters = radius_val
            boundary.is_active = True
            boundary.save()
        else:
            CampusBoundary.objects.create(
                name=(saved.school_name or "Campus Boundary")[:100],
                latitude=saved.latitude,
                longitude=saved.longitude,
                radius_meters=radius_val,
                is_active=True,
            )

        record_audit_log(
            request=request,
            table_name=saved._meta.db_table,
            record_id=saved.pk,
            action="UPDATE",
            changed_fields=list(serializer.validated_data.keys()),
        )
        data = dict(SchoolSettingsSerializer(saved, context={"request": request}).data)
        data.update(get_school_metadata(request, school=saved))
        return Response(data)

    def post(self, request):
        return self.put(request)

    def patch(self, request):
        return self.put(request)

    def delete(self, request):
        school = SchoolSettings.objects.first()
        field = request.query_params.get("field") or request.data.get("field")
        if school and field in ("school_logo", "left_logo", "right_logo"):
            setattr(school, field, None)
            school.save(update_fields=[field, "updated_at"])
            cache.delete_many([SCHOOL_META_CACHE_KEY, "reports:system_status"])
            record_audit_log(
                request=request,
                table_name=school._meta.db_table,
                record_id=school.pk,
                action="DELETE",
                changed_fields=[field],
            )
            return self.get(request)
        return Response(
            {"error": "Specify school_logo, left_logo, or right_logo."},
            status=status.HTTP_400_BAD_REQUEST,
        )

        weekend_rest = cache.get(WEEKEND_REST_CACHE_KEY)
        if weekend_rest is None:
            weekend_rest = True
        if isinstance(response.data, dict):
            response.data["weekend_rest_enabled"] = bool(weekend_rest)
        return response

    def partial_update_settings(self, request):
        val = request.data.get("weekend_rest_enabled")
        if val is not None:
            enabled = bool(val) if isinstance(val, bool) else str(val).strip().lower() in ("true", "1", "yes")
            cache.set(WEEKEND_REST_CACHE_KEY, enabled, timeout=None)
            return Response({"ok": True, "weekend_rest_enabled": enabled})
        current = cache.get(WEEKEND_REST_CACHE_KEY)
        return Response({"ok": True, "weekend_rest_enabled": True if current is None else bool(current)})

def get_students_for_user(*, user, search: str = "", section_id: int | None = None):
    role = get_user_role(user)
    qs = Student.objects.prefetch_related("enrollments__section", "enrollments__school_year").order_by(
        "last_name", "first_name"
    )
    if search:
        qs = qs.filter(Q(lrn__icontains=search) | Q(first_name__icontains=search) | Q(last_name__icontains=search))
    if section_id:
        qs = qs.filter(enrollments__section_id=section_id, enrollments__status__in=ACTIVE_ENROLLMENT_STATUSES)
    if role == "PARENT":
        return qs.filter(parent_user=user, is_active=True).distinct()
    if not is_school_leader(user):
        teacher = resolve_teacher_for_user(user)
        qs = qs.filter(id__in=get_teacher_student_ids(teacher)) if teacher else qs.none()
    return qs.distinct()

        raise AppError("FORBIDDEN", "Parent accounts cannot download school QR card batches.", 403)

    leader = is_school_leader(user)
    allowed_section_ids: set[int] | None = None
    if not leader:
        teacher = resolve_teacher_for_user(user)
        if not teacher:
            raise AppError("FORBIDDEN", "Only teachers and school leaders can access student QR cards.", 403)
        allowed_section_ids = get_teacher_section_ids(teacher)
        if section_id and section_id not in allowed_section_ids:
            raise AppError("FORBIDDEN", "You are not assigned to this section.", 403)

    qs = Enrollment.objects.filter(
        status__in=ACTIVE_ENROLLMENT_STATUSES,
        student__is_active=True,
    ).select_related("student", "section", "section__grade_level")

    active_sy = SchoolYear.objects.filter(is_active=True).first()
    if active_sy:
        qs = qs.filter(school_year=active_sy)
    if section_id:
        qs = qs.filter(section_id=section_id)
    elif allowed_section_ids is not None:
        qs = qs.filter(section_id__in=allowed_section_ids)

    return [
        {
            "student_id": enr.student.id,
            "lrn": enr.student.lrn,
            "full_name": enr.student.full_name,
            "grade_section": format_section_label(
                enr.section.grade_level.name if enr.section.grade_level else "", enr.section.name
            ),
            "qr_token": enr.student.qr_token,
            "qr_payload": sign_student_qr(enr.student.lrn, enr.student.qr_token),
            "photo_url": build_media_url(enr.student.photo, request),
        }
        for enr in qs.order_by("student__last_name", "student__first_name")[:500]
    ]


def get_parent_portal_data(*, user, request=None) -> dict:
    role = get_user_role(user)
    if role != "PARENT" and not is_school_leader(user):
        raise AppError("FORBIDDEN", "Only parent accounts can access the Parent Portal.", 403)

    today = timezone.localdate()
    children = list(Student.objects.filter(parent_user=user, is_active=True).order_by("first_name"))
    child_ids = [c.id for c in children]

    enr_qs = Enrollment.objects.filter(
        student_id__in=child_ids,
        status__in=ACTIVE_ENROLLMENT_STATUSES,
    ).select_related("section", "section__grade_level", "section__adviser").order_by("school_year__start_date", "id")
    active_sy = SchoolYear.objects.filter(is_active=True).first()
    if active_sy:
        enr_qs = enr_qs.filter(school_year=active_sy)
    enrollments = {e.student_id: e for e in enr_qs}
    daily_map = {
        d.student_id: d for d in DailyAttendance.objects.filter(student_id__in=child_ids, attendance_date=today)
    }

    scans_by_child: dict[int, list] = {}
    for log in (
        ClassAttendance.objects.filter(student_id__in=child_ids, attendance_date=today)
        .select_related("class_period__subject", "recorded_by")
        .order_by("-scanned_at")
    ):
        scans_by_child.setdefault(log.student_id, []).append({
            "id": log.id,
            "subject_code": log.class_period.subject.code if log.class_period else "CLASS",
            "subject_title": log.class_period.subject.title if log.class_period else "Class",
            "teacher_name": log.recorded_by.full_name if log.recorded_by else "",
            "status": log.status,
            "scanned_at": timezone.localtime(log.scanned_at).strftime("%I:%M %p"),
        })

    passes_by_child: dict[int, list] = {}
    for ep in ExitPass.objects.filter(student_id__in=child_ids).select_related("issued_by").order_by("-created_at")[:20]:
        passes_by_child.setdefault(ep.student_id, []).append({
            "id": ep.id,
            "pass_number": ep.pass_number,
            "reason": ep.reason,
            "status": ep.status,
            "valid_until": timezone.localtime(ep.valid_until).strftime("%I:%M %p"),
        })

    children_payload = []
    for child in children:
        enr = enrollments.get(child.id)
        sec = enr.section if enr else None
        daily = daily_map.get(child.id)
        children_payload.append({
            "id": child.id,
            "lrn": child.lrn,
            "full_name": child.full_name,
            "photo_url": build_media_url(child.photo, request),
            "section": format_section_label(sec.grade_level.name if (sec and sec.grade_level) else "", sec.name if sec else "Unassigned"),
            "adviser_name": sec.adviser.full_name if (sec and sec.adviser) else "",
            "today_status": daily.status if daily else "UNSCANNED",
            "subject_scans_today": scans_by_child.get(child.id, []),
            "exit_passes": passes_by_child.get(child.id, []),
        })

    alerts = list(
        Notification.objects.filter(Q(recipient=user) | Q(student_id__in=child_ids)).order_by("-created_at")[:50]
    )
    return {
        "parent_name": user.get_full_name() or user.username,
        "children_count": len(children_payload),
        "children": children_payload,
        "unread_notifications": sum(1 for a in alerts if not a.is_read),
        "notifications": [
            {
                "id": a.id,
                "title": a.title,
                "message": a.message,
                "category": a.category,
                "is_read": a.is_read,
                "created_at": timezone.localtime(a.created_at).strftime("%Y-%m-%d %I:%M %p"),
            }
            for a in alerts
        ],
    }

class StudentSerializer(serializers.ModelSerializer):
    full_name = serializers.CharField(read_only=True)
    photo = FlexibleImageField(required=False, allow_null=True)
    photo_url = serializers.SerializerMethodField()
    qr_payload = serializers.SerializerMethodField()
    section_id = serializers.IntegerField(required=False, allow_null=True, write_only=True)
    current_section = serializers.SerializerMethodField()
    current_section_id = serializers.SerializerMethodField()

    class Meta:
        model = Student
        fields = [
            "id", "lrn", "first_name", "middle_name", "last_name", "suffix",
            "full_name", "sex", "birthdate", "mother_tongue", "ethnic_group",
            "religion", "house_street_sitio", "barangay", "municipality_city",
            "province", "father_name", "mother_maiden_name", "guardian_name",
            "guardian_relationship", "parent_contact", "photo", "photo_url",
            "qr_token", "qr_payload", "is_active", "current_section", "current_section_id", "section_id",
        ]
        read_only_fields = ["id", "qr_token"]

    def validate_section_id(self, value):
        if value is not None and not Section.objects.filter(id=value).exists():
            raise serializers.ValidationError("Selected section does not exist.")
        return value

    def get_photo_url(self, obj):
        return build_media_url(obj.photo, self.context.get("request"))

    def get_qr_payload(self, obj) -> str:
        return sign_student_qr(obj.lrn, obj.qr_token)

    def _active_enrollment(self, obj):
        for enr in obj.enrollments.all():
            if enr.school_year and enr.school_year.is_active and enr.section:
                return enr
        return None

    def get_current_section(self, obj) -> str:
        enr = self._active_enrollment(obj)
        return enr.section.name if enr and enr.section else "Unassigned"

    def get_current_section_id(self, obj) -> int | None:
        enr = self._active_enrollment(obj)
        return enr.section_id if enr else None

    @transaction.atomic
    def create(self, validated_data):
        section_id = validated_data.pop("section_id", None)
        student = super().create(validated_data)
        sync_student_enrollment(student=student, section_id=section_id)
        return student

    @transaction.atomic
    def update(self, instance, validated_data):
        section_id = validated_data.pop("section_id", None)
        student = super().update(instance, validated_data)
        sync_student_enrollment(student=student, section_id=section_id)
        return student


class EnrollmentSerializer(serializers.ModelSerializer):
    student_name = serializers.CharField(source="student.full_name", read_only=True)
    lrn = serializers.CharField(source="student.lrn", read_only=True)
    section_name = serializers.CharField(source="section.name", read_only=True)

    class Meta:
        model = Enrollment
        fields = [
            "id", "student", "student_name", "lrn", "section",
            "section_name", "school_year", "enrollment_date", "status", "remarks",
        ]
        extra_kwargs = {
            "school_year": {"required": False},
            "enrollment_date": {"required": False},
        }

    def validate(self, attrs):
        sy = attrs.get("school_year") or (self.instance.school_year if self.instance else None)
        if not sy:
            sy = SchoolYear.objects.filter(is_active=True).first() or SchoolYear.objects.first()
            if not sy:
                raise serializers.ValidationError({"school_year": "No School Year is configured."})
            attrs["school_year"] = sy

        student = attrs.get("student") or (self.instance.student if self.instance else None)
        if student and sy:
            dup = Enrollment.objects.filter(student=student, school_year=sy)
            if self.instance:
                dup = dup.exclude(pk=self.instance.pk)
            if dup.exists():
                raise serializers.ValidationError(
                    {"student": "This student is already enrolled for the selected school year."}
                )

        if not self.instance and not attrs.get("enrollment_date"):
            attrs["enrollment_date"] = timezone.localdate()
        return attrs

    def create(self, validated_data):
        validated_data.setdefault("enrollment_date", timezone.localdate())
        return super().create(validated_data)

class StudentViewSet(AuditedModelViewSet):
    serializer_class = StudentSerializer
    permission_classes = [permissions.IsAuthenticated, IsLeadershipOrReadOnly]

    def get_queryset(self):
        sec_id = self.request.query_params.get("section_id") or self.request.query_params.get("section")
        return get_students_for_user(
            user=self.request.user,
            search=self.request.query_params.get("search", "").strip(),
            section_id=int(sec_id) if (sec_id and str(sec_id).isdigit()) else None,
        )

    @action(detail=False, methods=["get"], url_path="metrics")
    def metrics(self, request):
        base_qs = self.get_queryset()
        agg = base_qs.aggregate(
            total=Count("id", distinct=True),
            active=Count("id", filter=Q(is_active=True), distinct=True),
            male=Count("id", filter=Q(is_active=True, sex="Male"), distinct=True),
            female=Count("id", filter=Q(is_active=True, sex="Female"), distinct=True),
            enrolled=Count(
                "id",
                filter=Q(is_active=True, enrollments__status__in=["ENROLLED", "TRANSFERRED_IN"]),
                distinct=True,
            ),
            paired=Count(
                "id",
                filter=Q(is_active=True, qr_token__isnull=False) & ~Q(qr_token=""),
                distinct=True,
            ),
        )
        total = agg["total"] or 0
        active = agg["active"] or 0
        enrolled = agg["enrolled"] or 0
        total_sections = get_sections_for_user(request.user).count()

        return Response({
            "total_students": active,
            "total": total,
            "active": active,
            "paired": agg["paired"] or 0,
            "total_sections": total_sections,
            "male": agg["male"] or 0,
            "female": agg["female"] or 0,
            "enrolled": enrolled,
            "unassigned": max(0, active - enrolled),
        })

    @action(detail=False, methods=["get"], url_path="dropdown")
    def dropdown(self, request):
        qs = self.get_queryset().filter(is_active=True).only(
            "id", "lrn", "first_name", "middle_name", "last_name", "suffix", "sex"
        )[:500]
        return Response([
            {
                "id": st.id,
                "lrn": st.lrn,
                "first_name": st.first_name,
                "last_name": st.last_name,
                "full_name": st.full_name,
                "sex": st.sex,
            }
            for st in qs
        ])

    @action(detail=False, methods=["get"], url_path="qr-cards")
    def qr_cards(self, request):
        sec_id = request.query_params.get("section_id")
        cards = get_signed_qr_cards(
            user=request.user,
            section_id=int(sec_id) if (sec_id and str(sec_id).isdigit()) else None,
            request=request,
        )
        return Response({"cards": cards, "total": len(cards)})

def get_assigned_classes_with_rosters(
    *,
    user,
    view_all: bool = False,
    include_students: bool = True,
    request=None,
) -> list[dict]:
    from django.db.models import Count

    teacher = resolve_teacher_for_user(user)
    leader = is_school_leader(user)
    today = timezone.localdate()

    periods_qs = ClassPeriod.objects.none()
    extra_sections_qs = Section.objects.none()

    if teacher and not view_all:
        own_periods = ClassPeriod.objects.filter(
            Q(teacher=teacher) | Q(section__adviser=teacher),
            is_active=True,
        ).distinct()
        own_advisory = Section.objects.filter(adviser=teacher, is_active=True)

        if own_periods.exists() or own_advisory.exists():
            periods_qs = own_periods
            extra_sections_qs = own_advisory
        elif leader:
            periods_qs = ClassPeriod.objects.filter(is_active=True)
            extra_sections_qs = Section.objects.filter(is_active=True)
    elif leader:
        periods_qs = ClassPeriod.objects.filter(is_active=True)
        extra_sections_qs = Section.objects.filter(is_active=True)

    periods = list(
        periods_qs.select_related("section", "section__grade_level", "subject", "teacher").order_by("start_time")
    )
    covered_section_ids = {p.section_id for p in periods if p.section_id}
    extra_sections = list(
        extra_sections_qs.exclude(id__in=covered_section_ids)
        .select_related("grade_level")
        .order_by("grade_level__level_order", "name")
    )

    all_section_ids = covered_section_ids | {s.id for s in extra_sections}
    if not all_section_ids:
        return []

    if not include_students:
        counts_by_sec = dict(
            Enrollment.objects.filter(
                section_id__in=all_section_ids,
                status__in=ACTIVE_ENROLLMENT_STATUSES,
                student__is_active=True,
            )
            .values("section_id")
            .annotate(cnt=Count("id"))
            .values_list("section_id", "cnt")
        )
        results = []
        for p in periods:
            if not p.section:
                continue
            start_s = str(p.start_time)[:5] if p.start_time else ""
            end_s = str(p.end_time)[:5] if p.end_time else ""
            cnt = counts_by_sec.get(p.section_id, 0)
            results.append({
                "id": p.id,
                "schedule_id": p.id,
                "section_id": p.section_id,
                "name": p.section.name,
                "section_name": p.section.name,
                "grade_level": p.section.grade_level.name if p.section.grade_level else "",
                "subject_code": p.subject.code if p.subject else "",
                "subject_title": p.subject.title if p.subject else "",
                "room": p.section.room_number or p.room_number,
                "start_time": start_s,
                "end_time": end_s,
                "time_display": f"{start_s} - {end_s}".strip(" -"),
                "total_students": cnt,
                "students": [],
                "is_adviser": bool(teacher and p.section.adviser_id == teacher.id),
            })
        for sec in extra_sections:
            results.append({
                "id": -sec.id,
                "schedule_id": None,
                "section_id": sec.id,
                "name": sec.name,
                "section_name": sec.name,
                "grade_level": sec.grade_level.name if sec.grade_level else "",
                "subject_code": "ADV",
                "subject_title": "Advisory Homeroom",
                "room": sec.room_number or "",
                "start_time": "",
                "end_time": "",
                "time_display": "Homeroom",
                "total_students": counts_by_sec.get(sec.id, 0),
                "students": [],
                "is_adviser": bool(teacher and sec.adviser_id == teacher.id),
            })
        return results

    enrollments = (

            attendance_date=today,
            student_id__in=all_student_ids,
        ).values_list("student_id", "status")
    )

    results = []
    for p in periods:

            continue
        roster = format_roster_items(
            students_by_sec.get(p.section_id, []),
            request=request,
            period_logs=period_logs,

        )
        start_s = str(p.start_time)[:5] if p.start_time else ""
        end_s = str(p.end_time)[:5] if p.end_time else ""
        results.append({

            "students": roster,

        })

    for sec in extra_sections:
        roster = format_roster_items(students_by_sec.get(sec.id, []), request=request, daily_logs=daily_logs)
        results.append({
            "id": -sec.id,

            "section_id": sec.id,

            "room": sec.room_number or "",
            "start_time": "",

            "time_display": "Homeroom",
            "total_students": len(roster),
            "students": roster,
            "is_adviser": bool(teacher and sec.adviser_id == teacher.id),
        })

    return results

def get_adviser_matrix(*, user, section_id: int | None = None, request=None) -> dict:
    today = timezone.localdate()
    now_time = timezone.localtime().time()
    teacher = resolve_teacher_for_user(user)
    leader = is_school_leader(user)

    advisory_ids = (
        set(Section.objects.filter(adviser=teacher, is_active=True).values_list("id", flat=True))
        if teacher
        else set()
    )

    if leader:
        sections_qs = (
            Section.objects.filter(is_active=True)
            .select_related("grade_level")
            .order_by("grade_level__level_order", "name")
        )
    elif teacher:
        sections_qs = (
            Section.objects.filter(id__in=get_teacher_section_ids(teacher), is_active=True)
            .select_related("grade_level")
            .order_by("grade_level__level_order", "name")
        )
    else:
        sections_qs = Section.objects.none()

    available = [
        {
            "id": s.id,
            "name": s.name,
            "grade_level": s.grade_level.name if s.grade_level else "",
            "is_adviser": s.id in advisory_ids,
        }
        for s in sections_qs
    ]

    target = (
        sections_qs.filter(id=section_id).first()
        if section_id
        else (sections_qs.filter(id__in=advisory_ids).first() or sections_qs.first())
    )
    if not target:
        return {
            "has_advisory_class": False,
            "is_adviser": False,
            "available_sections": available,
            "schedules": [],
            "matrix": [],
        }

    all_periods = list(
        ClassPeriod.objects.filter(section=target, is_active=True)
        .select_related("subject", "teacher")
        .prefetch_related("days")
        .order_by("start_time")
    )
    is_holiday = SchoolHoliday.objects.filter(holiday_date=today).exists()
    periods = [p for p in all_periods if is_period_scheduled_on(p, today)] or all_periods
    students = [
        e.student
        for e in Enrollment.objects.filter(
            section=target, status__in=ACTIVE_ENROLLMENT_STATUSES, student__is_active=True
        )
        .select_related("student")
        .order_by("student__last_name", "student__first_name")
    ]
    student_ids = [s.id for s in students]

    log_map: dict[tuple[int, int], tuple[str, str]] = {}
    first_seen_map: dict[int, str] = {}
    started_period_ids: set[int] = set()

    on_campus_ids = set(
        DailyAttendance.objects.filter(
            attendance_date=today,
            student_id__in=student_ids,
            status__in=["PRESENT", "LATE"],
        ).values_list("student_id", flat=True)
    )

    attendance_rows = (
        ClassAttendance.objects.filter(attendance_date=today, student_id__in=student_ids)
        .values("student_id", "class_period_id", "status", "scanned_at", "class_period__subject__code")
        .order_by("scanned_at")
    )

    for row in attendance_rows:
        started_period_ids.add(row["class_period_id"])
        t_str = timezone.localtime(row["scanned_at"]).strftime("%I:%M %p") if row["scanned_at"] else ""
        st_code = row["status"]
        log_map[(row["student_id"], row["class_period_id"])] = (st_code, t_str)

        if st_code in ("PRESENT", "LATE"):
            on_campus_ids.add(row["student_id"])
            if row["student_id"] not in first_seen_map:
                subj_code = row["class_period__subject__code"] or "CLASS"
                first_seen_map[row["student_id"]] = f"{t_str} ({subj_code})" if t_str else subj_code

    matrix = []
    total_periods = len(periods)

    for idx, st in enumerate(students, start=1):
        attended = cutting = 0
        in_campus = st.id in on_campus_ids
        subj_cells = []

        for p in periods:
            rec = log_map.get((st.id, p.id))
            if rec:
                st_status, scan_t = rec
                if st_status in ("PRESENT", "LATE", "EXCUSED"):
                    attended += 1
                elif st_status in ("ABSENT", "CUTTING") and in_campus:
                    st_status = "CUTTING"
                    cutting += 1
            else:
                scan_t = ""
                grace_passed = False
                if p.start_time and not is_holiday and is_period_scheduled_on(p, today):
                    grace_dt = datetime.combine(today, p.start_time) + timedelta(minutes=15)
                    grace_passed = now_time >= grace_dt.time()
                if in_campus and not is_holiday and (p.id in started_period_ids or grace_passed):
                    st_status = "CUTTING"
                    cutting += 1
                else:
                    st_status = "UNSCANNED"

            subj_cells.append({
                "schedule_id": p.id,
                "subject_code": p.subject.code if p.subject else "SUB",
                "status": st_status,
                "time": scan_t,
            })

        campus_status = first_seen_map.get(st.id) or ("Present Today" if in_campus else "Not Seen Yet")

        matrix.append({
            "number": idx,
            "student_id": st.id,
            "student_name": st.full_name,
            "lrn": st.lrn,
            "photo_url": build_media_url(st.photo, request),
            "campus_status": campus_status,
            "presence_state": "ACTIVE" if in_campus else "NONE",
            "attended_count": attended,
            "cutting_count": cutting,
            "total_subjects": total_periods,
            "compliance_rate": round((attended / total_periods) * 100) if total_periods else 0,
            "subjects": subj_cells,
        })

    return {
        "has_advisory_class": True,
        "is_adviser": target.id in advisory_ids,
        "section_id": target.id,
        "section_name": target.name,
        "grade_level": target.grade_level.name if target.grade_level else "",
        "total_students": len(students),
        "available_sections": available,
        "schedules": [
            {
                "schedule_id": p.id,
                "subject_code": p.subject.code if p.subject else "",
                "subject_name": p.subject.title if p.subject else "",
                "faculty_name": p.teacher.full_name if p.teacher else "Unassigned",
                "time_window": f"{str(p.start_time)[:5]} - {str(p.end_time)[:5]}" if p.start_time else "",
            }
            for p in periods
        ],
        "matrix": matrix,
    }

def get_exit_passes_for_user(*, user, status_filter: str = "ALL", search: str = ""):
    role = get_user_role(user)
    qs = ExitPass.objects.select_related("student", "teacher", "issued_by").order_by("-created_at")

    if status_filter and status_filter.upper() != "ALL":
        clean_st = status_filter.upper()
        if clean_st == "DECLINED":
            qs = qs.filter(status__in=["REVOKED", "DECLINED"])
        else:
            qs = qs.filter(status=clean_st)
    if search:
        qs = qs.filter(
            Q(pass_number__icontains=search)
            | Q(student__first_name__icontains=search)
            | Q(student__last_name__icontains=search)
            | Q(teacher__first_name__icontains=search)
            | Q(teacher__last_name__icontains=search)
            | Q(reason__icontains=search)
        )
    if role == "PARENT":
        return qs.filter(student__parent_user=user)
    if not is_school_leader(user):
        teacher = resolve_teacher_for_user(user)
        if not teacher:
            return qs.none()
        qs = qs.filter(
            Q(teacher=teacher)
            | Q(issued_by=teacher)
            | Q(student_id__in=get_teacher_student_ids(teacher))
        )
    return qs

class ExitPassSerializer(serializers.ModelSerializer):
    bearer_type = serializers.SerializerMethodField()
    student_name = serializers.CharField(source="student.full_name", read_only=True, default="")
    student_lrn = serializers.CharField(source="student.lrn", read_only=True, default="")
    teacher_name = serializers.CharField(source="teacher.full_name", read_only=True, default="")
    faculty_name = serializers.CharField(source="teacher.full_name", read_only=True, default="")
    faculty_emp_id = serializers.CharField(source="teacher.employee_id", read_only=True, default="")
    adviser_name = serializers.CharField(source="issued_by.full_name", read_only=True, default="School Staff")
    valid_to = serializers.DateTimeField(source="valid_until", required=False)

    class Meta:
        model = ExitPass
        fields = [
            "id", "pass_number", "bearer_type",
            "student", "student_name", "student_lrn",
            "teacher", "teacher_name", "faculty_name", "faculty_emp_id",
            "issued_by", "adviser_name", "pass_type", "reason",
            "valid_from", "valid_until", "valid_to", "status", "created_at",
        ]
        read_only_fields = ["id", "pass_number", "issued_by", "status", "created_at"]
        extra_kwargs = {
            "valid_from": {"required": False},
            "valid_until": {"required": False},
        }

    def get_bearer_type(self, obj) -> str:
        return "STUDENT" if obj.student_id else "FACULTY"

    def to_representation(self, instance):
        data = super().to_representation(instance)
        if data.get("status") == "REVOKED":
            data["status"] = "DECLINED"
        return data

    def validate(self, attrs):
        now = timezone.now()
        attrs.setdefault("valid_from", now)
        attrs.setdefault("valid_until", attrs["valid_from"] + timedelta(hours=4))
        if attrs["valid_until"] <= attrs["valid_from"]:
            raise serializers.ValidationError({"valid_until": "Return time must be after departure time."})
        return attrs

    def create(self, validated_data):
        request = self.context["request"]
        user = request.user
        role = get_user_role(user)
        leader = is_school_leader(user)

        if role == "PARENT":
            raise AppError("FORBIDDEN", "Parent accounts cannot issue exit passes.", 403)

        own_teacher = resolve_teacher_for_user(user)
        issuer = own_teacher or (TeacherProfile.objects.filter(is_active=True).first() if leader else None)
        if not issuer:
            raise AppError(
                "NO_TEACHER_PROFILE",
                "A linked teacher profile is required to issue an exit pass.",
                403,
            )

        target_student = validated_data.get("student")
        target_teacher = validated_data.get("teacher")

        if target_student:
            if not leader and target_student.id not in get_teacher_student_ids(own_teacher):
                raise AppError("FORBIDDEN", "You can only issue exit passes for students in your assigned classes.", 403)
            validated_data["teacher"] = None
        else:
            if not leader:
                validated_data["teacher"] = own_teacher
            elif not target_teacher:
                validated_data["teacher"] = issuer

        is_student_pass = bool(validated_data.get("student"))
        initial_status = "ACTIVE" if (leader or is_student_pass) else "PENDING"
        pass_code = f"EP-{timezone.localdate().strftime('%Y%m%d')}-{uuid.uuid4().hex[:6].upper()}"

        pass_obj = ExitPass.objects.create(
            **validated_data,
            pass_number=pass_code,
            issued_by=issuer,
            status=initial_status,
        )
        if pass_obj.student:
            notify_student_parent(
                student=pass_obj.student,
                title=f"Exit Pass Issued: {pass_obj.student.first_name}",
                message=f"Exit pass #{pass_code} issued for {pass_obj.student.full_name} ({pass_obj.reason}).",
                category="GATE_PASS",
            )
        return pass_obj

ACTIVE_ENROLLMENT_STATUSES = ["ENROLLED", "TRANSFERRED_IN"]
DAY_PRESETS = {"MON-FRI": {1, 2, 3, 4, 5}, "MWF": {1, 3, 5}, "TTH": {2, 4}, "SAT": {6}}
DAY_TOKENS = {"MON": 1, "TUE": 2, "WED": 3, "THU": 4, "FRI": 5, "SAT": 6, "SUN": 7}


def is_period_scheduled_on(period: ClassPeriod, target_date) -> bool:
    day_num = target_date.isoweekday()
    explicit_days = list(period.days.values_list("day_of_week", flat=True))
    if explicit_days:
        return day_num in explicit_days
    clean = str(period.days_of_week or "MON-FRI").upper().strip()
    if clean in DAY_PRESETS:
        return day_num in DAY_PRESETS[clean]
    parsed = {DAY_TOKENS[p.strip()] for p in clean.split(",") if p.strip() in DAY_TOKENS}
    return day_num in (parsed or {1, 2, 3, 4, 5})


def _resolve_active_enrollment(student: Student):
    qs = Enrollment.objects.filter(
        student=student, status__in=ACTIVE_ENROLLMENT_STATUSES
    ).select_related("section", "section__grade_level", "section__adviser")
    active_sy = SchoolYear.objects.filter(is_active=True).first()
    if active_sy:
        enr = qs.filter(school_year=active_sy).order_by("-id").first()
        if enr:
            return enr
    return qs.order_by("-id").first()

def _recompute_daily_attendance(student: Student, section, target_date, remarks: str = "") -> None:
    if not section:
        return
    statuses = list(
        ClassAttendance.objects.filter(student=student, attendance_date=target_date).values_list("status", flat=True)
    )
    if not statuses:
        DailyAttendance.objects.filter(student=student, attendance_date=target_date).delete()
        return

    status_set = set(statuses)
    if "CUTTING" in status_set:
        daily_status = "CUTTING"
    elif "PRESENT" in status_set:
        daily_status = "PRESENT"
    elif "LATE" in status_set:
        daily_status = "LATE"
    elif "EXCUSED" in status_set:
        daily_status = "EXCUSED"
    else:
        daily_status = "ABSENT"

    DailyAttendance.objects.update_or_create(
        student=student,
        attendance_date=target_date,
        defaults={"section": section, "status": daily_status, "remarks": remarks[:255]},
    )


def determine_arrival_status(period: ClassPeriod | None, scanned_at, grace_minutes: int = 15) -> str:
    if not period or not period.start_time:
        return "PRESENT"
    local_dt = timezone.localtime(scanned_at)
    cutoff = timezone.make_aware(
        datetime.combine(local_dt.date(), period.start_time),
        timezone.get_current_timezone(),
    ) + timedelta(minutes=grace_minutes)
    return "LATE" if local_dt > cutoff else "PRESENT"

    active_period = None
    if period_id:
        requested_period = ClassPeriod.objects.select_related("subject", "teacher", "section").filter(
            id=period_id, is_active=True
        ).first()
        if not requested_period:
            raise AppError("NO_SCHEDULE", "Selected class period was not found or is inactive.", 404)
        if requested_period.section_id != section.id:
            raise AppError(
                "WRONG_SECTION",
                f"{student.full_name} belongs to {section_label}, not the selected class period.",
                400,
            )
        active_period = requested_period
    elif teacher:
        active_period = period_qs.filter(teacher=teacher).first()

    is_adviser = bool(teacher and section.adviser_id == teacher.id)
    if not (leader or is_adviser or (active_period and teacher and active_period.teacher_id == teacher.id)):
        raise AppError("FORBIDDEN_SECTION", f"You are not assigned to {section_label}.", 403)

    target_period = active_period or period_qs.order_by("start_time").first()
    if not target_period:
        raise AppError("NO_SCHEDULE", f"No active class period found for {section_label}.", 400)

    recorder = _resolve_recorder_teacher(target_period, teacher, section=section)

    now = timezone.now()
    today = timezone.localdate()

    is_holiday = SchoolHoliday.objects.filter(holiday_date=today).exists()
    missed_count = 0
    if not is_holiday and target_period.start_time:
        earlier_periods = [
            p
            for p in ClassPeriod.objects.filter(
                section=section, is_active=True, start_time__lt=target_period.start_time
            ).prefetch_related("days")
            if is_period_scheduled_on(p, today)
        ]
        if earlier_periods:
            attended_period_ids = set(
                ClassAttendance.objects.filter(
                    student=student,
                    attendance_date=today,
                    class_period_id__in=[p.id for p in earlier_periods],
                    status__in=["PRESENT", "LATE", "EXCUSED"],
                ).values_list("class_period_id", flat=True)
            )
            missed_count = sum(1 for p in earlier_periods if p.id not in attended_period_ids)

    note = (
        f"CUTTING ALERT: Missed {missed_count} earlier period(s) today."
        if missed_count > 0
        else "Classroom QR attendance verified."
    )

    arrival_status = determine_arrival_status(target_period, now)
    record, created = ClassAttendance.objects.get_or_create(
        student=student,
        class_period=target_period,
        attendance_date=today,
        defaults={
            "recorded_by": recorder,
            "scanned_at": now,
            "scan_method": "QR",
            "status": arrival_status,
        },
    )

    if created:
        _recompute_daily_attendance(student, section, today, remarks=note)
        subject_name = target_period.subject.title if target_period.subject else "Class"
        time_display = timezone.localtime(now).strftime("%I:%M %p")
        notify_student_parent(
            student=student,
            title=f"Cutting Alert: {student.first_name}" if missed_count > 0 else f"Attendance: {student.first_name} ({arrival_status})",
            message=(
                f"{student.full_name} scanned into {subject_name} at {time_display}, missing {missed_count} earlier period(s)."
                if missed_count > 0
                else f"{student.full_name} was marked {arrival_status} in {subject_name} at {time_display}."
            ),
            category="CUTTING_ALERT" if missed_count > 0 else "ATTENDANCE",
        )

    return {
        "ok": True,
        "success": True,
        "is_duplicate": not created,
        "student_id": student.id,
        "student_name": student.full_name,
        "lrn": student.lrn,
        "section": section_label,
        "status": record.status,
        "scanned_at": timezone.localtime(record.scanned_at).strftime("%I:%M:%S %p"),
        "attendance_note": note,
        "photo_url": build_media_url(student.photo, request),
    }


@transaction.atomic
def sync_offline_qr_scans(*, user, raw_scans: list[dict]) -> dict:
    teacher = resolve_teacher_for_user(user)
    leader = is_school_leader(user)
    if not teacher and not leader:
        raise AppError("FORBIDDEN", "Only teachers and administrators can sync offline scans.", 403)

    allowed_section_ids = get_teacher_section_ids(teacher) if (teacher and not leader) else None

    parsed = []
    tokens = set()
    for item in raw_scans[:500]:
        if not isinstance(item, dict):
            continue
        qr_data = verify_student_qr(item.get("qr_token") or item.get("code") or "")
        if not qr_data:
            continue
        tokens.add(qr_data.qr_token)
        tokens.add(qr_data.lrn)
        parsed.append({
            "qr": qr_data,
            "period_id": item.get("schedule_id") or item.get("period_id"),
            "scanned_at": parse_datetime(str(item.get("scanned_at", ""))) or timezone.now(),
        })

    students = Student.objects.filter(Q(qr_token__in=tokens) | Q(lrn__in=tokens), is_active=True)
    student_lookup = {}
    for st in students:
        student_lookup[st.qr_token] = st
        student_lookup[st.lrn] = st

    enr_qs = Enrollment.objects.filter(
        student__in=students, status__in=ACTIVE_ENROLLMENT_STATUSES
    ).select_related("section", "section__adviser").order_by("school_year__start_date", "id")
    active_sy = SchoolYear.objects.filter(is_active=True).first()
    if active_sy:
        enr_qs = enr_qs.filter(school_year=active_sy)
    enrollments = {e.student_id: e.section for e in enr_qs}

    synced = duplicates = 0
    rejected = len(raw_scans) - len(parsed)

    for entry in parsed:
        st = student_lookup.get(entry["qr"].qr_token) or student_lookup.get(entry["qr"].lrn)
        sec = enrollments.get(st.id) if st else None
        if not st or not sec:
            rejected += 1
            continue

        if allowed_section_ids is not None and sec.id not in allowed_section_ids:
            rejected += 1
            continue

        pid = entry["period_id"]
        if pid and str(pid).isdigit():
            period = ClassPeriod.objects.select_related("teacher").filter(
                id=int(pid), section=sec, is_active=True
            ).first()
        else:
            period = (
                ClassPeriod.objects.select_related("teacher")
                .filter(section=sec, teacher=teacher, is_active=True)
                .order_by("start_time")
                .first()
                if teacher
                else None
            ) or ClassPeriod.objects.select_related("teacher").filter(section=sec, is_active=True).order_by("start_time").first()

        if not period:
            rejected += 1
            continue

        is_adviser = bool(teacher and sec.adviser_id == teacher.id)
        if not (leader or is_adviser or (teacher and period.teacher_id == teacher.id)):
            rejected += 1
            continue

        recorder = period.teacher or teacher or sec.adviser
        if not recorder:
            rejected += 1
            continue

        scan_dt = entry["scanned_at"]
        if timezone.is_naive(scan_dt):
            scan_dt = timezone.make_aware(scan_dt, timezone.get_current_timezone())
        scan_date = timezone.localtime(scan_dt).date()
        status_val = determine_arrival_status(period, scan_dt)

        _, created = ClassAttendance.objects.get_or_create(
            student=st,
            class_period=period,
            attendance_date=scan_date,
            defaults={
                "recorded_by": recorder,
                "scanned_at": scan_dt,
                "scan_method": "QR",
                "status": status_val,
            },
        )
        if not created:
            duplicates += 1
            continue

        _recompute_daily_attendance(st, sec, scan_date, remarks="Synced from offline QR queue.")
        synced += 1

    return {
        "ok": True,
        "success": True,
        "synced_count": synced,
        "duplicate_count": duplicates,
        "rejected_count": rejected,
    }


@transaction.atomic
def save_manual_attendance(
    *, user, student_id: int, status_code: str, period_id: int | None = None, date_str: str | None = None
) -> ClassAttendance:
    clean_status = status_code.upper().strip()
    if clean_status not in {"PRESENT", "LATE", "CUTTING", "ABSENT", "EXCUSED"}:
        raise AppError("INVALID_STATUS", "Invalid attendance status.", 400)

    student = Student.objects.filter(id=student_id, is_active=True).first()
    if not student:
        raise AppError("STUDENT_NOT_FOUND", "Student not found.", 404)

    teacher = resolve_teacher_for_user(user)
    leader = is_school_leader(user)
    if not leader and (not teacher or student.id not in get_teacher_student_ids(teacher)):
        raise AppError("FORBIDDEN", "Access denied for this student.", 403)

    enr = _resolve_active_enrollment(student)
    sec = enr.section if enr else None
    if not sec:
        raise AppError("NOT_ENROLLED", f"{student.full_name} has no active section.", 400)

    if period_id:
        period = ClassPeriod.objects.select_related("teacher").filter(
            id=period_id, section=sec, is_active=True
        ).first()
        if not period:
            raise AppError("NO_SCHEDULE", "Class period does not belong to this student's section.", 400)
    else:
        period = (
            ClassPeriod.objects.select_related("teacher")
            .filter(section=sec, teacher=teacher, is_active=True)
            .order_by("start_time")
            .first()
            if teacher
            else None
        ) or ClassPeriod.objects.select_related("teacher").filter(section=sec, is_active=True).order_by("start_time").first()

    if not period:
        raise AppError("NO_SCHEDULE", "No active class period found.", 400)

    is_adviser = bool(teacher and sec.adviser_id == teacher.id)
    if not (leader or is_adviser or (teacher and period.teacher_id == teacher.id)):
        raise AppError("FORBIDDEN", "You can only record attendance for your own assigned periods or advisory section.", 403)

    recorder = _resolve_recorder_teacher(period, teacher, section=sec)

    target_date = timezone.localdate()
    if date_str:
        try:
            target_date = datetime.strptime(date_str, "%Y-%m-%d").date()
        except ValueError:
            raise AppError("INVALID_DATE", "attendance_date must be YYYY-MM-DD.", 400)

    log, _ = ClassAttendance.objects.update_or_create(
        student=student,
        class_period=period,
        attendance_date=target_date,
        defaults={
            "recorded_by": recorder,
            "status": clean_status,
            "scan_method": "MANUAL",
            "scanned_at": timezone.now(),
        },
    )
    _recompute_daily_attendance(student, sec, target_date, remarks="Manual teacher update.")
    return log


@transaction.atomic
def delete_attendance_record(*, user, record_id: int) -> None:
    record = (
        ClassAttendance.objects.select_related("student", "class_period", "class_period__section")
        .filter(id=record_id)
        .first()
    )
    if not record:
        raise AppError("NOT_FOUND", "Attendance record not found.", 404)

    leader = is_school_leader(user)
    if not leader:
        teacher = resolve_teacher_for_user(user)
        period = record.class_period
        sec = period.section if period else None
        can_delete = bool(
            teacher
            and (
                record.recorded_by_id == teacher.id
                or (period and period.teacher_id == teacher.id)
                or (sec and sec.adviser_id == teacher.id)
            )
        )
        if not can_delete:
            raise AppError("FORBIDDEN", "You do not have permission to delete this attendance record.", 403)

    student = record.student
    sec = record.class_period.section if record.class_period else None
    att_date = record.attendance_date
    record.delete()
    _recompute_daily_attendance(student, sec, att_date, remarks="Updated after record removal.")


def change_exit_pass_status(*, user, exit_pass: ExitPass, next_status: str) -> ExitPass:
    role = get_user_role(user)
    leader = is_school_leader(user)

    if next_status in ("ACTIVE", "REVOKED") and not leader:
        raise AppError("FORBIDDEN", "Only school leadership can approve or decline passes.", 403)

    if next_status == "ACTIVE" and exit_pass.status != "PENDING":
        raise AppError("INVALID_PASS_STATE", f"Only pending passes can be approved (current: {exit_pass.status}).", 400)

    if next_status == "REVOKED" and exit_pass.status not in ("PENDING", "ACTIVE"):
        raise AppError("INVALID_PASS_STATE", f"Cannot decline a pass that is already {exit_pass.status}.", 400)

    if next_status == "USED":
        if role == "PARENT" or not (leader or resolve_teacher_for_user(user)):
            raise AppError("FORBIDDEN", "Only school staff or administrators can mark a pass as used.", 403)
        if exit_pass.status != "ACTIVE":
            raise AppError("INVALID_PASS_STATE", f"Pass is currently {exit_pass.status}.", 400)
        now = timezone.now()
        if exit_pass.valid_from and now < exit_pass.valid_from:
            raise AppError("PASS_NOT_YET_VALID", "Exit pass is not valid yet.", 400)
        if exit_pass.valid_until and now > exit_pass.valid_until:
            exit_pass.status = "EXPIRED"
            exit_pass.save(update_fields=["status", "updated_at"])
            raise AppError("PASS_EXPIRED", "Exit pass has expired.", 400)

    with transaction.atomic():
        exit_pass.status = next_status
        exit_pass.save(update_fields=["status", "updated_at"])

        if next_status == "USED" and exit_pass.student:
            notify_student_parent(
                student=exit_pass.student,
                title=f"Authorized Campus Exit: {exit_pass.student.first_name}",
                message=f"{exit_pass.student.full_name} exited campus using approved pass #{exit_pass.pass_number}.",
                category="GATE_PASS",
            )
    return exit_pass

class ClassAttendanceView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    def get(self, request):
        date_str = request.query_params.get("date")
        target_date = timezone.localdate()
        if date_str:
            try:
                target_date = datetime.strptime(date_str, "%Y-%m-%d").date()
            except ValueError:
                pass
        sec_id = request.query_params.get("section_id")
        per_id = request.query_params.get("schedule_id") or request.query_params.get("period_id")
        records = get_attendance_logs(
            user=request.user,
            target_date=target_date,
            section_id=int(sec_id) if (sec_id and str(sec_id).isdigit()) else None,
            period_id=int(per_id) if (per_id and str(per_id).isdigit()) else None,
        )
        return Response({"records": records, "total": len(records)})

    def post(self, request):
        student_id = request.data.get("student_id")
        if not student_id or not str(student_id).isdigit():
            raise AppError("INVALID_STUDENT_ID", "A valid numeric student_id is required.", 400)
        per_id = request.data.get("schedule_id") or request.data.get("period_id")
        log = save_manual_attendance(
            user=request.user,
            student_id=int(student_id),
            status_code=str(request.data.get("status", "PRESENT")),
            period_id=int(per_id) if (per_id and str(per_id).isdigit()) else None,
            date_str=request.data.get("attendance_date"),
        )
        return Response({"ok": True, "success": True, "id": log.id, "status": log.status}, status=status.HTTP_201_CREATED)

    def put(self, request):
        rec_id = request.data.get("id") or request.query_params.get("id")
        if rec_id is not None and str(rec_id).strip() != "":
            if not str(rec_id).isdigit():
                raise AppError("INVALID_ID", "A valid numeric attendance record id is required.", 400)
            existing = ClassAttendance.objects.filter(id=int(rec_id)).first()
            if not existing:
                raise AppError("NOT_FOUND", "Attendance record not found.", 404)
            log = save_manual_attendance(
                user=request.user,
                student_id=existing.student_id,
                status_code=str(request.data.get("status", existing.status)),
                period_id=existing.class_period_id,
                date_str=str(existing.attendance_date),
            )
            return Response({"ok": True, "success": True, "id": log.id, "status": log.status})
        return self.post(request)

    def patch(self, request):
        return self.put(request)

    def delete(self, request):
        rec_id = request.data.get("id") or request.query_params.get("id")
        if not rec_id or not str(rec_id).isdigit():
            raise AppError("INVALID_ID", "A valid numeric attendance record id is required.", 400)
        delete_attendance_record(user=request.user, record_id=int(rec_id))
        return Response({"ok": True, "success": True, "message": "Attendance record removed."})

ONLINE_WINDOW_MINUTES = 5
ACTIVE_ENROLLMENT_STATUSES = ["ENROLLED", "TRANSFERRED_IN"]


def get_active_campus_coords() -> tuple[float, float, int, str]:
    boundary = CampusBoundary.objects.filter(is_active=True).only(
        "latitude", "longitude", "radius_meters", "name"
    ).first()
    school = SchoolSettings.objects.only(
        "latitude", "longitude", "geofence_radius_meters", "school_name"
    ).first()

    lat = (
        float(boundary.latitude)
        if (boundary and boundary.latitude is not None)
        else (float(school.latitude) if (school and school.latitude is not None) else 8.4858)
    )
    lon = (
        float(boundary.longitude)
        if (boundary and boundary.longitude is not None)
        else (float(school.longitude) if (school and school.longitude is not None) else 124.6567)
    )
    radius = (
        int(boundary.radius_meters)
        if (boundary and boundary.radius_meters)
        else (int(school.geofence_radius_meters) if (school and school.geofence_radius_meters) else 150)
    )
    name = boundary.name if boundary else (school.school_name if school else "Campus Boundary")
    return lat, lon, radius, name


def get_campus_boundary_overview(*, user, request=None) -> dict:
    now_dt = timezone.now()
    today = timezone.localdate()
    today_start = timezone.make_aware(datetime.combine(today, datetime.min.time()))
    online_cutoff = now_dt - timedelta(minutes=ONLINE_WINDOW_MINUTES)

    lat, lon, radius, name = get_active_campus_coords()
    leader = is_school_leader(user)

    if leader:
        teachers = list(
            TeacherProfile.objects.filter(is_active=True)
            .only(
                "id",
                "employee_id",
                "first_name",
                "middle_name",
                "last_name",
                "suffix",
                "department",
                "position",
                "bound_device_id",
                "device_model",
                "photo",
            )
            .order_by("last_name", "first_name")
        )
        base_logs_qs = TeacherLocationLog.objects.filter(recorded_at__gte=today_start)
        logs_qs = base_logs_qs.select_related("teacher").order_by("-recorded_at")[:300]
        alerts_qs = (
            BoundaryAlert.objects.filter(incident_date=today)
            .select_related("teacher")
            .order_by("-created_at")[:100]
        )
    else:
        own_teacher = resolve_teacher_for_user(user)
        teachers = [own_teacher] if own_teacher else []
        base_logs_qs = (
            TeacherLocationLog.objects.filter(teacher=own_teacher, recorded_at__gte=today_start)
            if own_teacher
            else TeacherLocationLog.objects.none()
        )
        logs_qs = base_logs_qs.select_related("teacher").order_by("-recorded_at")[:60]
        alerts_qs = (
            BoundaryAlert.objects.filter(teacher=own_teacher, incident_date=today)
            .select_related("teacher")
            .order_by("-created_at")
            if own_teacher
            else BoundaryAlert.objects.none()
        )

    raw_logs = list(logs_qs)
    latest_by_teacher: dict[int, TeacherLocationLog] = {}
    logs: list[dict] = []

    for item in raw_logs:
        latest_by_teacher.setdefault(item.teacher_id, item)
        local_rec = timezone.localtime(item.recorded_at)
        dist = calculate_distance_meters(item.latitude, item.longitude, lat, lon)
        logs.append(
            {
                "id": item.id,
                "teacher_id": item.teacher_id,
                "faculty_id": item.teacher_id,
                "employee_id": item.teacher.employee_id if item.teacher else "",
                "teacher_name": item.teacher.full_name if item.teacher else "Unknown",
                "faculty_name": item.teacher.full_name if item.teacher else "Unknown",
                "latitude": float(item.latitude),
                "longitude": float(item.longitude),
                "distance": dist if dist != float("inf") else None,
                "accuracy": float(getattr(item, "accuracy_meters", 0) or 0) or None,
                "battery_level": item.battery_level,
                "battery": item.battery_level,
                "is_inside_geofence": item.is_inside_geofence,
                "is_inside": item.is_inside_geofence,
                "verified_inside": item.is_inside_geofence,
                "recorded_at": local_rec.strftime("%I:%M:%S %p"),
                "time_only": local_rec.strftime("%I:%M:%S %p"),
                "date_only": local_rec.strftime("%Y-%m-%d"),
                "timestamp": local_rec.strftime("%Y-%m-%d %I:%M:%S %p"),
            }
        )

    # Ensure teachers whose latest ping today was older than the 300-row global slice aren't dropped
    missing_teacher_ids = [t.id for t in teachers if t.id not in latest_by_teacher]
    if missing_teacher_ids:
        latest_ids = list(
            base_logs_qs.filter(teacher_id__in=missing_teacher_ids)
            .values("teacher_id")
            .annotate(latest_id=Max("id"))
            .values_list("latest_id", flat=True)
        )
        if latest_ids:
            for item in TeacherLocationLog.objects.filter(id__in=latest_ids):
                latest_by_teacher[item.teacher_id] = item

    faculty_list: list[dict] = []
    online_count = 0
    verified_inside_count = 0
    outside_teacher_count = 0

    for t in teachers:
        latest = latest_by_teacher.get(t.id)
        is_online = bool(latest and latest.recorded_at >= online_cutoff)
        inside = bool(latest and latest.is_inside_geofence)
        dist = (
            calculate_distance_meters(latest.latitude, latest.longitude, lat, lon)
            if latest
            else None
        )
        if dist == float("inf"):
            dist = None

        if is_online:
            online_count += 1
        if latest:
            if inside:
                verified_inside_count += 1
            else:
                outside_teacher_count += 1

        if latest:
            if inside:
                status_label = "VERIFIED_INSIDE" if is_online else "OFFLINE"
            else:
                status_label = "PERIMETER_BREACH" if is_online else "OFF_CAMPUS"
        else:
            status_label = "OFFLINE"

        local_seen = timezone.localtime(latest.recorded_at) if latest else None
        bound_id = getattr(t, "bound_device_id", None)

        faculty_list.append(
            {
                "id": t.id,
                "teacher_id": t.id,
                "faculty_id": t.id,
                "employee_id": t.employee_id,
                "full_name": t.full_name,
                "faculty_name": t.full_name,
                "teacher_name": t.full_name,
                "department": t.department or "",
                "position": t.position or "",
                "photo_url": build_media_url(t.photo, request),
                "is_online": is_online,
                "status": status_label,
                "is_inside_geofence": inside,
                "is_inside": inside,
                "verified_inside": inside,
                "latitude": float(latest.latitude) if latest else None,
                "longitude": float(latest.longitude) if latest else None,
                "distance": dist,
                "battery_level": latest.battery_level if latest else None,
                "battery": latest.battery_level if latest else None,
                "last_seen": local_seen.strftime("%I:%M:%S %p") if local_seen else "No signal today",
                "recorded_at": local_seen.strftime("%I:%M:%S %p") if local_seen else None,
                "bound_device_id": bound_id,
                "is_device_bound": bool(bound_id),
                "device_model": getattr(t, "device_model", None) or "",
            }
        )

    offline_count = max(len(teachers) - online_count, 0)

    boundary_alerts = [
        {
            "id": a.id,
            "teacher_id": a.teacher_id,
            "faculty_id": a.teacher_id,
            "employee_id": a.teacher.employee_id if a.teacher else "",
            "teacher_name": a.teacher.full_name if a.teacher else "Unknown",
            "faculty_name": a.teacher.full_name if a.teacher else "Unknown",
            "trigger_reason": a.trigger_reason,
            "reason": a.trigger_reason,
            "status": a.status,
            "created_at": timezone.localtime(a.created_at).strftime("%I:%M %p"),
            "date_only": timezone.localtime(a.created_at).strftime("%Y-%m-%d"),
            "time_only": timezone.localtime(a.created_at).strftime("%I:%M:%S %p"),
            "timestamp": timezone.localtime(a.created_at).strftime("%Y-%m-%d %I:%M:%S %p"),
        }
        for a in alerts_qs
    ]

    return {
        "name": name,
        "latitude": lat,
        "longitude": lon,
        "radius_meters": radius,
        "geofence_radius_meters": radius,
        "is_active": True,
        "inside_count": verified_inside_count,
        "verified_inside": verified_inside_count,
        "outside_count": outside_teacher_count,
        "online_users_count": online_count,
        "offline_users_count": offline_count,
        "incidents_today": len(boundary_alerts),
        "faculty_list": faculty_list,
        "heartbeats": logs,
        "logs": logs,
        "boundary_alerts": boundary_alerts,
        "recent_breaches": boundary_alerts,
    }


def get_campus_audit_history(
    *,
    user,
    date_from: str = "",
    date_to: str = "",
    faculty_id: int | None = None,
    status_filter: str = "",
    search: str = "",
    request=None,
) -> dict:
    lat, lon, _, _ = get_active_campus_coords()
    leader = is_school_leader(user)
    own_teacher = None if leader else resolve_teacher_for_user(user)

    parsed_from = parse_date(str(date_from or "").strip())
    parsed_to = parse_date(str(date_to or "").strip())
    clean_status = str(status_filter or "").strip().upper()
    clean_search = str(search or "").strip().lower()[:80]

    hb_qs = TeacherLocationLog.objects.select_related("teacher").order_by("-recorded_at")
    alert_qs = BoundaryAlert.objects.select_related("teacher").order_by("-created_at")

    if not leader:
        if own_teacher:
            hb_qs = hb_qs.filter(teacher=own_teacher)
            alert_qs = alert_qs.filter(teacher=own_teacher)
        else:
            hb_qs = TeacherLocationLog.objects.none()
            alert_qs = BoundaryAlert.objects.none()
    elif faculty_id:
        hb_qs = hb_qs.filter(teacher_id=faculty_id)
        alert_qs = alert_qs.filter(teacher_id=faculty_id)

    if parsed_from:
        start_dt = timezone.make_aware(datetime.combine(parsed_from, datetime.min.time()))
        hb_qs = hb_qs.filter(recorded_at__gte=start_dt)
        alert_qs = alert_qs.filter(incident_date__gte=parsed_from)
    if parsed_to:
        end_dt = timezone.make_aware(datetime.combine(parsed_to + timedelta(days=1), datetime.min.time()))
        hb_qs = hb_qs.filter(recorded_at__lt=end_dt)
        alert_qs = alert_qs.filter(incident_date__lte=parsed_to)
    if clean_status == "INSIDE":
        hb_qs = hb_qs.filter(is_inside_geofence=True)
        alert_qs = BoundaryAlert.objects.none()
    elif clean_status in ("OUTSIDE", "BREACH"):
        hb_qs = hb_qs.filter(is_inside_geofence=False)

        AuditLog.objects.select_related("performed_by")
        .filter(table_name__in=tracked_tables)

    )

    if parsed_from:
        audit_qs = audit_qs.filter(created_at__gte=timezone.make_aware(datetime.combine(parsed_from, datetime.min.time())))
    if parsed_to:
        audit_qs = audit_qs.filter(created_at__lt=timezone.make_aware(datetime.combine(parsed_to + timedelta(days=1), datetime.min.time())))

    records: list[dict] = []
    heartbeats_out: list[dict] = []
    alerts_out: list[dict] = []

        local_dt = timezone.localtime(a.created_at)
        t_name = a.teacher.full_name if a.teacher else "Unknown"
        emp_id = a.teacher.employee_id if a.teacher else ""
        row = {
            "id": f"alert-{a.id}",
            "category": "BREACH",
            "teacher_id": a.teacher_id,
            "faculty_id": a.teacher_id,
            "user": t_name,
            "faculty_name": t_name,
            "employee_id": emp_id,
            "position": a.teacher.position if a.teacher else "",
            "action": "BOUNDARY_ALERT",
            "status": a.status,
            "verdict": "Outside School Grounds",
            "details": a.trigger_reason,
            "is_inside_geofence": False,
            "is_inside": False,
            "distance": None,
            "battery": None,
            "battery_level": None,
            "ip_address": "",
            "date_only": local_dt.strftime("%Y-%m-%d"),
            "time_only": local_dt.strftime("%I:%M:%S %p"),
            "timestamp": local_dt.strftime("%Y-%m-%d %I:%M:%S %p"),
        }
        if not clean_search or clean_search in f"{t_name} {emp_id} {a.trigger_reason}".lower():
            alerts_out.append(row)

    for item in hb_qs[:200]:

        t_name = item.teacher.full_name if item.teacher else "Unknown"
        emp_id = item.teacher.employee_id if item.teacher else ""
        dist = calculate_distance_meters(item.latitude, item.longitude, lat, lon)

            dist = None
        row = {
            "id": f"hb-{item.id}",
            "category": "HEARTBEAT",
            "teacher_id": item.teacher_id,
            "faculty_id": item.teacher_id,
            "user": t_name,
            "faculty_name": t_name,
            "employee_id": emp_id,
            "position": item.teacher.position if item.teacher else "",
            "action": "INSIDE_CAMPUS" if item.is_inside_geofence else "OUTSIDE_CAMPUS",
            "status": "INSIDE" if item.is_inside_geofence else "OUTSIDE",

def get_live_campus_anomalies(*, user) -> list[dict]:
    from apps.attendance.services import is_period_scheduled_on

    leader = is_school_leader(user)
    if not (leader or get_user_role(user) == "FACULTY"):
        return []

    today = timezone.localdate()
    if SchoolHoliday.objects.filter(holiday_date=today).exists():
        return []

    now_dt = timezone.now()
    now_time = timezone.localtime(now_dt).time()
    today_start = timezone.make_aware(datetime.combine(today, datetime.min.time()))

    student_passes = set(
        ExitPass.objects.filter(
            student__isnull=False,
            status__in=["ACTIVE", "USED"],
            valid_from__lte=now_dt,
            valid_until__gte=now_dt,
        ).values_list("student_id", flat=True)
    )
    teacher_passes = set(
        ExitPass.objects.filter(
            teacher__isnull=False,
            status__in=["ACTIVE", "USED"],
            valid_from__lte=now_dt,
            valid_until__gte=now_dt,
        ).values_list("teacher_id", flat=True)
    )
    present_today = set(
        DailyAttendance.objects.filter(
            attendance_date=today, status__in=["PRESENT", "LATE"]
        ).values_list("student_id", flat=True)
    )

    periods_qs = (
        ClassPeriod.objects.select_related("section", "subject", "teacher")
        .prefetch_related("days")
        .filter(is_active=True, start_time__lte=now_time)
    )
    if not leader:
        own_teacher = resolve_teacher_for_user(user)
        if not own_teacher:
            return []
        periods_qs = periods_qs.filter(section_id__in=get_teacher_section_ids(own_teacher))

    periods = [p for p in periods_qs if is_period_scheduled_on(p, today)]
    period_ids = [p.id for p in periods]
    section_ids = {p.section_id for p in periods if p.section_id}

    verified_pairs = set(
        ClassAttendance.objects.filter(
            attendance_date=today,
            class_period_id__in=period_ids,
            status__in=["PRESENT", "LATE", "EXCUSED"],
        ).values_list("student_id", "class_period_id")
    )
    started_periods = {pid for _, pid in verified_pairs}

    rosters: dict[int, list] = {}
    for enr in Enrollment.objects.filter(
        section_id__in=section_ids, status__in=ACTIVE_ENROLLMENT_STATUSES, student__is_active=True
    ).select_related("student"):
        rosters.setdefault(enr.section_id, []).append(enr.student)

    anomalies = []
    time_str = timezone.localtime(now_dt).strftime("%I:%M %p")

    for period in periods:
        if not period.section or period.id not in started_periods:
            continue
        for st in rosters.get(period.section_id, []):
            if st.id in present_today and (st.id, period.id) not in verified_pairs and st.id not in student_passes:
                subj = period.subject.code if period.subject else "CLASS"
                anomalies.append(
                    {
                        "category": "STUDENT",
                        "anomaly_type": "CUTTING_CLASS",
                        "severity": "HIGH",
                        "person_id": st.id,
                        "lrn": st.lrn,
                        "name": st.full_name,
                        "section": period.section.name,
                        "subject": subj,
                        "details": f"Present earlier today but missing from {subj} without an exit pass.",
                        "detected_at": time_str,
                    }
                )

    if leader:
        active_periods = [p for p in periods if p.teacher_id and p.end_time and p.end_time >= now_time]
        teacher_ids = {p.teacher_id for p in active_periods}
        latest_logs = {}
        for log in TeacherLocationLog.objects.filter(
            teacher_id__in=teacher_ids, recorded_at__gte=today_start
        ).order_by("teacher_id", "-recorded_at"):
            latest_logs.setdefault(log.teacher_id, log)

        for period in active_periods:
            t = period.teacher
            if not t or t.id in teacher_passes:
                continue
            latest = latest_logs.get(t.id)
            if latest and not latest.is_inside_geofence:
                subj = period.subject.code if period.subject else "CLASS"
                anomalies.append(
                    {
                        "category": "FACULTY",
                        "anomaly_type": "OUTSIDE_GEOFENCE",
                        "severity": "HIGH",
                        "person_id": t.id,
                        "employee_id": t.employee_id,
                        "name": t.full_name,
                        "section": period.section.name if period.section else "",
                        "subject": subj,
                        "details": f"Outside campus boundary during {subj} without an exit pass.",
                        "detected_at": timezone.localtime(latest.recorded_at).strftime("%I:%M %p"),
                    }
                )

    return anomalies


def get_emergency_roll_call(*, request=None) -> dict:
    today = timezone.localdate()
    today_start = timezone.make_aware(datetime.combine(today, datetime.min.time()))

    departed_ids = set(
        ExitPass.objects.filter(
            student__isnull=False, valid_from__date=today, status="USED"
        ).values_list("student_id", flat=True)
    )
    present_rows = (
        DailyAttendance.objects.filter(attendance_date=today, status__in=["PRESENT", "LATE"])
        .exclude(student_id__in=departed_ids)
        .select_related("student", "section", "section__grade_level")
        .order_by("section__grade_level__level_order", "section__name", "student__last_name")
    )

    students = [
        {
            "student_id": row.student_id,
            "lrn": row.student.lrn,
            "full_name": row.student.full_name,
            "section": format_section_label(
                row.section.grade_level.name if (row.section and row.section.grade_level) else "",
                row.section.name if row.section else "",
            ),
            "parent_contact": row.student.parent_contact,
            "photo_url": build_media_url(row.student.photo, request),
        }
        for row in present_rows
    ]

    latest_inside_by_teacher: dict[int, bool] = {}
    for log in TeacherLocationLog.objects.filter(recorded_at__gte=today_start).order_by("-recorded_at"):
        latest_inside_by_teacher.setdefault(log.teacher_id, log.is_inside_geofence)

    inside_teacher_ids = [tid for tid, is_in in latest_inside_by_teacher.items() if is_in]
    teachers = [
        {
            "teacher_id": t.id,
            "faculty_id": t.id,
            "employee_id": t.employee_id,
            "full_name": t.full_name,
            "department": t.department,
        }
        for t in TeacherProfile.objects.filter(id__in=inside_teacher_ids, is_active=True).order_by("last_name")
    ]

    return {
        "timestamp": timezone.localtime().strftime("%Y-%m-%d %I:%M:%S %p"),
        "total_people_on_campus": len(students) + len(teachers),
        "students_count": len(students),
        "faculty_count": len(teachers),
        "students": students,
        "faculty": teachers,
    }

@transaction.atomic
def update_campus_boundary(*, request, name: str, latitude: float, longitude: float, radius_meters: int) -> CampusBoundary:
    if not (20 <= radius_meters <= 5000):
        raise AppError("INVALID_RADIUS", "Campus radius must be between 20 and 5000 meters.", 400)

    clean_name = (str(name or "").strip() or "Campus Boundary")[:100]
    boundary = CampusBoundary.objects.filter(is_active=True).first() or CampusBoundary.objects.first()
    if boundary:
        boundary.name = clean_name
        boundary.latitude = latitude
        boundary.longitude = longitude
        boundary.radius_meters = radius_meters
        boundary.is_active = True
        boundary.save()
    else:
        boundary = CampusBoundary.objects.create(
            name=clean_name,
            latitude=latitude,
            longitude=longitude,
            radius_meters=radius_meters,
            is_active=True,
        )

    school = SchoolSettings.objects.first()
    if school:
        school.latitude = latitude
        school.longitude = longitude
        school.geofence_radius_meters = radius_meters
        school.save(update_fields=["latitude", "longitude", "geofence_radius_meters", "updated_at"])

    record_audit_log(
        request=request,
        table_name="geofence_settings",
        record_id=boundary.id,
        action="UPDATE",
        new_data={"latitude": latitude, "longitude": longitude, "radius_meters": radius_meters},
    )
    return boundary


@transaction.atomic
def record_teacher_check_in(*, user, validated_data: dict, header_device_id: str = "") -> dict:
    teacher = resolve_teacher_for_user(user)
    if not teacher:
        raise AppError("NO_TEACHER_PROFILE", "No teacher profile is linked to this account.", 403)

    if validated_data.get("face_verified") is False or validated_data.get("liveness_passed") is False:
        raise AppError("FACE_CHECK_FAILED", "Mobile face or liveness verification failed.", 403)

    device_id = str(validated_data.get("client_device_id") or header_device_id or "").strip()[:128]
    if teacher.bound_device_id:
        if not device_id or teacher.bound_device_id != device_id:
            raise AppError(
                "DEVICE_MISMATCH",
                "Account is bound to another phone. Request a device reset from an administrator.",
                403,
            )
    elif device_id:
        if TeacherProfile.objects.filter(bound_device_id=device_id).exclude(id=teacher.id).exists():
            raise AppError(
                "DEVICE_ALREADY_BOUND",
                "This phone is already bound to another faculty account. Request a device reset from an administrator.",
                403,
            )
        teacher.bound_device_id = device_id
        teacher.device_bound_at = timezone.now()
        teacher.save(update_fields=["bound_device_id", "device_bound_at"])

    campus_lat, campus_lon, radius, _ = get_active_campus_coords()
    gps = verify_campus_gps(
        latitude=validated_data["latitude"],
        longitude=validated_data["longitude"],
        campus_lat=campus_lat,
        campus_lon=campus_lon,
        radius_meters=radius,
        accuracy_meters=validated_data.get("accuracy"),
        is_mock_location=validated_data.get("is_mock_location", False),
    )
    if not gps["is_valid"]:
        raise AppError("INVALID_GPS", gps["reason"], 400)

    now = timezone.now()
    battery_val = validated_data.get("battery_level")
    if battery_val is None:
        battery_val = 100

    log = TeacherLocationLog.objects.create(
        teacher=teacher,
        latitude=validated_data["latitude"],
        longitude=validated_data["longitude"],
        battery_level=battery_val,
        is_inside_geofence=gps["is_inside"],
        recorded_at=now,
    )

    if gps["is_inside"]:
        local_now = timezone.localtime(now)
        today = local_now.date()
        is_holiday = SchoolHoliday.objects.filter(holiday_date=today).exists()
        if not is_holiday:
            cutoff_dt = local_now - timedelta(minutes=15)
            cutoff_time = cutoff_dt.time() if cutoff_dt.date() == today else time(0, 0)
            candidates = (
                ClassPeriod.objects.filter(
                    teacher=teacher,
                    is_active=True,
                    start_time__lte=cutoff_time,
                    end_time__gte=local_now.time(),
                )
                .exclude(attendance_records__attendance_date=today)
                .select_related("subject")
                .prefetch_related("days")
            )
            unlogged_period = next((p for p in candidates if is_period_scheduled_on(p, today)), None)
            if unlogged_period:
                subj_code = unlogged_period.subject.code if unlogged_period.subject else "CLASS"
                reason = f"On campus via mobile check-in, but no QR scans recorded for {subj_code}."
                if not BoundaryAlert.objects.filter(
                    teacher=teacher,
                    incident_date=today,
                    trigger_reason=reason,
                ).exists():
                    BoundaryAlert.objects.create(
                        teacher=teacher,
                        incident_date=today,
                        trigger_reason=reason,
                    )

    return {
        "ok": True,
        "success": True,
        "heartbeat_id": log.id,
        "is_inside_geofence": gps["is_inside"],
        "distance_meters": gps["distance_meters"],
        "allowed_radius": gps["allowed_radius"],
        "message": gps["reason"],
    }

class CampusBoundaryView(APIView):
    permission_classes = [permissions.IsAuthenticated, IsLeadershipOrReadOnly]

    def get(self, request):
        return Response(get_campus_boundary_overview(user=request.user, request=request))

    def put(self, request):
        payload = (
            {k: request.data.get(k) for k in request.data.keys()}
            if hasattr(request.data, "keys")
            else dict(request.data)
        )
        if "geofence_radius_meters" in payload and "radius_meters" not in payload:
            payload["radius_meters"] = payload["geofence_radius_meters"]
        serializer = CampusBoundaryInputSerializer(data=payload)
        serializer.is_valid(raise_exception=True)
        update_campus_boundary(request=request, **serializer.validated_data)
        return Response(get_campus_boundary_overview(user=request.user, request=request))

    def post(self, request):
        return self.put(request)

    def patch(self, request):
        return self.put(request)

def resolve_default_teacher_for_student(student: Student) -> TeacherProfile | None:
    enr = (
        Enrollment.objects.filter(student=student, status__in=["ENROLLED", "TRANSFERRED_IN"])
        .select_related("section", "section__adviser")
        .order_by("-id")
        .first()
    )
    if enr and enr.section:
        if enr.section.adviser:
            return enr.section.adviser
        period = (
            ClassPeriod.objects.filter(section=enr.section, is_active=True, teacher__isnull=False)
            .select_related("teacher")
            .first()
        )
        if period and period.teacher:
            return period.teacher
    return TeacherProfile.objects.filter(is_active=True).order_by("id").first()


def authorize_chat_thread(*, user, student_id: int, teacher_id: int | None = None):
    student = Student.objects.select_related("parent_user").filter(id=student_id, is_active=True).first()
    if not student:
        raise AppError("STUDENT_NOT_FOUND", "Student not found.", 404)

    role = get_user_role(user)
    if role == "PARENT":
        if student.parent_user_id != user.id:
            raise AppError("FORBIDDEN", "This student is not linked to your parent account.", 403)
        teacher = TeacherProfile.objects.filter(id=teacher_id, is_active=True).first() if teacher_id else None
        teacher = teacher or resolve_default_teacher_for_student(student)
        if not teacher:
            raise AppError("TEACHER_NOT_FOUND", "Assigned teacher not found.", 404)
        return student, teacher, user

    if not student.parent_user:
        raise AppError("NO_PARENT_LINKED", "This student has no linked parent account yet.", 400)

    leader = is_school_leader(user)
    own_teacher = resolve_teacher_for_user(user)
    if not leader:
        if not own_teacher or student.id not in get_teacher_student_ids(own_teacher):
            raise AppError("FORBIDDEN", "Student is not in your assigned classes.", 403)
        if teacher_id and teacher_id != own_teacher.id:
            raise AppError("FORBIDDEN", "You can only access your own conversation threads.", 403)
        return student, own_teacher, student.parent_user

    teacher = TeacherProfile.objects.filter(id=teacher_id, is_active=True).first() if teacher_id else None
    teacher = teacher or own_teacher or resolve_default_teacher_for_student(student)
    if not teacher:
        raise AppError("TEACHER_NOT_FOUND", "Assigned teacher not found.", 404)

    return student, teacher, student.parent_user


def get_chat_threads_for_user(*, user, request=None) -> list[dict]:
    role = get_user_role(user)
    own_teacher = resolve_teacher_for_user(user) if role != "PARENT" else None

    if role == "PARENT":
        base_qs = Student.objects.filter(parent_user=user, is_active=True)
    else:
        if is_school_leader(user) and not own_teacher:
            base_qs = Student.objects.filter(is_active=True, parent_user__isnull=False)
        elif own_teacher:
            base_qs = Student.objects.filter(
                id__in=get_teacher_student_ids(own_teacher),
                is_active=True,
                parent_user__isnull=False,
            )
        else:
            base_qs = Student.objects.none()

    students = list(base_qs.select_related("parent_user").order_by("last_name", "first_name")[:150])
    student_ids = [s.id for s in students]

    enrollments = {
        e.student_id: e.section
        for e in Enrollment.objects.filter(
            student_id__in=student_ids, status__in=["ENROLLED", "TRANSFERRED_IN"]
        )
        .select_related("section", "section__grade_level", "section__adviser")
        .order_by("school_year__start_date", "id")
    }

    # Resolve exact (student_id, teacher_id) pair per thread before loading previews/unreads
    thread_pairs: list[tuple[Student, object, TeacherProfile]] = []
    pair_set: set[tuple[int, int]] = set()
    for st in students:
        sec = enrollments.get(st.id)
        teacher = own_teacher or (sec.adviser if (sec and sec.adviser) else None) or resolve_default_teacher_for_student(st)
        if not teacher or not st.parent_user:
            continue
        thread_pairs.append((st, sec, teacher))
        pair_set.add((st.id, teacher.id))

    latest_by_pair: dict[tuple[int, int], tuple[str, object]] = {}
    unread_by_pair: dict[tuple[int, int], int] = {}
    if student_ids:
        msg_qs = ChatMessage.objects.filter(student_id__in=student_ids).order_by("-created_at")
        if own_teacher:
            msg_qs = msg_qs.filter(teacher=own_teacher)
        for msg in msg_qs.only("student_id", "teacher_id", "sender_id", "body", "is_read", "created_at"):
            pair = (msg.student_id, msg.teacher_id)
            if pair not in pair_set:
                continue
            if pair not in latest_by_pair:
                latest_by_pair[pair] = (msg.body, msg.created_at)
            if not msg.is_read and msg.sender_id != user.id:
                unread_by_pair[pair] = unread_by_pair.get(pair, 0) + 1

    threads_with_sort = []
    for st, sec, teacher in thread_pairs:
        pair = (st.id, teacher.id)
        last_body, last_dt = latest_by_pair.get(pair, ("", None))
        unread_count = unread_by_pair.get(pair, 0)

        section_label = format_section_label(sec.grade_level.name if (sec and sec.grade_level) else "", sec.name if sec else "")
        parent_name = st.guardian_name or st.mother_maiden_name or st.father_name or st.parent_user.get_full_name() or st.parent_user.username
        photo_field = teacher.photo if role == "PARENT" else st.photo

        threads_with_sort.append(
            (
                (unread_count > 0, last_dt.timestamp() if last_dt else 0.0),
                {
                    "thread_key": f"{st.id}-{teacher.id}",
                    "student_id": st.id,
                    "student_name": st.full_name,
                    "student_lrn": st.lrn,
                    "section": section_label,
                    "faculty_id": teacher.id,
                    "teacher_id": teacher.id,
                    "faculty_name": teacher.full_name,
                    "teacher_name": teacher.full_name,
                    "parent_id": st.parent_user_id,
                    "parent_name": parent_name,
                    "photo_url": build_media_url(photo_field, request),
                    "last_message": last_body or "Tap to start conversation",
                    "last_time": timezone.localtime(last_dt).strftime("%I:%M %p") if last_dt else "",
                    "unread_count": unread_count,
                },
            )
        )

    threads_with_sort.sort(key=lambda item: item[0], reverse=True)
    return [item[1] for item in threads_with_sort]


def get_thread_messages(*, user, student_id: int, teacher_id: int | None = None) -> dict:
    student, teacher, parent_user = authorize_chat_thread(
        user=user, student_id=student_id, teacher_id=teacher_id
    )
    base_qs = ChatMessage.objects.filter(
        student=student, teacher=teacher, parent=parent_user
    )
    base_qs.filter(is_read=False).exclude(sender=user).update(is_read=True)

    recent_messages = list(base_qs.select_related("sender").order_by("-created_at")[:200])
    recent_messages.reverse()

    return {
        "student_id": student.id,
        "student_name": student.full_name,
        "teacher_id": teacher.id,
        "faculty_id": teacher.id,
        "teacher_name": teacher.full_name,
        "faculty_name": teacher.full_name,
        "messages": [
            {
                "id": m.id,
                "sender_id": m.sender_id,
                "sender_name": m.sender.get_full_name() or m.sender.username,
                "is_mine": m.sender_id == user.id,
                "body": m.body,
                "is_read": m.is_read,
                "created_at": timezone.localtime(m.created_at).strftime("%Y-%m-%d %I:%M %p"),
                "time_only": timezone.localtime(m.created_at).strftime("%I:%M %p"),
            }
            for m in recent_messages
        ],
    }

from django.db import transaction
from common.exceptions import AppError
from apps.accounts.models import UserProfile
from .models import ChatMessage, Notification
from .tasks import deliver_expo_push_task


def send_user_notification(
    *,
    recipient,
    title: str,
    message: str,
    category: str = "ATTENDANCE",
    student=None,
) -> Notification | None:
    if not recipient or not recipient.is_active:
        return None

    notif = Notification.objects.create(
        recipient=recipient,
        student=student,
        title=title[:150],
        message=message,
        category=category,
    )

    profile = getattr(recipient, "profile", None) or UserProfile.objects.filter(user=recipient).first()
    push_token = profile.push_token if profile else None
    if push_token and str(push_token).startswith(("ExponentPushToken", "ExpoPushToken")):
        transaction.on_commit(
            lambda: deliver_expo_push_task.delay(
                notification_id=notif.id,
                push_token=push_token,
                title=title[:150],
                body=message[:250],
            )
        )
    return notif


def notify_student_parent(*, student, title: str, message: str, category: str = "ATTENDANCE"):
    if not student:
        return None

    parent_user = student.parent_user
    if not parent_user and student.parent_contact:
        matched = (
            UserProfile.objects.filter(
                role="PARENT",
                phone_number=student.parent_contact,
                user__is_active=True,
            )
            .select_related("user")
            .first()
        )
        if matched:
            parent_user = matched.user
            student.parent_user = parent_user
            student.save(update_fields=["parent_user"])

    if not parent_user:
        return None

    return send_user_notification(
        recipient=parent_user,
        title=title,
        message=message,
        category=category,
        student=student,
    )


def mark_notifications_read(*, user, notification_ids: list[int] | None = None) -> int:
    qs = Notification.objects.filter(recipient=user, is_read=False)
    if notification_ids:
        qs = qs.filter(id__in=notification_ids)
    return qs.update(is_read=True)


@transaction.atomic
def create_chat_message(*, sender, student, teacher, parent_user, body: str) -> ChatMessage:
    clean_body = str(body or "").strip()
    if not clean_body:
        raise AppError("EMPTY_MESSAGE", "Message text cannot be empty.", 400)
    if len(clean_body) > 1000:
        raise AppError("MESSAGE_TOO_LONG", "Message cannot exceed 1000 characters.", 400)

    msg = ChatMessage.objects.create(
        student=student,
        teacher=teacher,
        parent=parent_user,
        sender=sender,
        body=clean_body,
    )

    teacher_user = teacher.account.user if (hasattr(teacher, "account") and teacher.account) else None
    recipient = teacher_user if sender.id == parent_user.id else parent_user
    sender_name = sender.get_full_name() or sender.username

    if recipient:
        send_user_notification(
            recipient=recipient,
            title=f"New Message ({student.first_name}): {sender_name}",
            message=clean_body[:180],
            category="ANNOUNCEMENT",
            student=student,
        )
    return msg


def retry_pending_push_notifications(limit: int = 100) -> int:
    pending = (
        Notification.objects.filter(is_pushed=False)
        .select_related("recipient__profile")
        .order_by("-created_at")[:limit]
    )
    queued = 0
    for notif in pending:
        profile = getattr(notif.recipient, "profile", None)
        push_token = profile.push_token if profile else None
        if push_token and str(push_token).startswith(("ExponentPushToken", "ExpoPushToken")):
            deliver_expo_push_task.delay(
                notification_id=notif.id,
                push_token=push_token,
                title=notif.title[:150],
                body=notif.message[:250],
            )
            queued += 1
    return queued


@transaction.atomic
def update_chat_message(*, user, message_id: int, new_body: str) -> ChatMessage:
    clean_body = str(new_body or "").strip()
    if not clean_body or len(clean_body) > 1000:
        raise AppError("INVALID_MESSAGE", "Message must be between 1 and 1000 characters.", 400)

    msg = ChatMessage.objects.filter(id=message_id).first()
    if not msg:
        raise AppError("MESSAGE_NOT_FOUND", "Message not found.", 404)
    if msg.sender_id != user.id:
        raise AppError("FORBIDDEN", "You can only edit your own messages.", 403)

    msg.body = clean_body
    msg.save(update_fields=["body"])
    return msg


@transaction.atomic
def delete_chat_message(*, user, message_id: int) -> None:
    msg = ChatMessage.objects.filter(id=message_id).first()
    if not msg:
        raise AppError("MESSAGE_NOT_FOUND", "Message not found.", 404)
    if msg.sender_id != user.id and not user.is_superuser:
        raise AppError("FORBIDDEN", "You can only delete your own messages.", 403)
    msg.delete()

import logging
import requests
from celery import shared_task
from .models import Notification

logger = logging.getLogger(__name__)
EXPO_PUSH_URL = "https://exp.host/--/api/v2/push/send"


@shared_task(bind=True, max_retries=2, default_retry_delay=5)
def deliver_expo_push_task(self, notification_id: int, push_token: str, title: str, body: str):
    if not str(push_token or "").startswith(("ExponentPushToken", "ExpoPushToken")):
        return
    try:
        resp = requests.post(
            EXPO_PUSH_URL,
            json={"to": push_token, "sound": "default", "title": title[:150], "body": body[:250]},
            timeout=5,
        )
        resp.raise_for_status()
        body_json = resp.json() if resp.content else {}
        ticket = body_json.get("data") if isinstance(body_json, dict) else None
        if isinstance(ticket, list):
            ticket = ticket[0] if ticket else {}
        if isinstance(ticket, dict) and ticket.get("status") == "error":
            logger.warning("Expo push rejected for notification %s: %s", notification_id, ticket)
            return
        Notification.objects.filter(id=notification_id).update(is_pushed=True)
    except requests.RequestException as exc:
        logger.warning("Push retry for notification %s: %s", notification_id, exc)
        raise self.retry(exc=exc)

class NotificationViewSet(viewsets.ModelViewSet):
    serializer_class = NotificationSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        return Notification.objects.select_related("student").filter(
            recipient=self.request.user
        ).order_by("-created_at")

    def create(self, request, *args, **kwargs):
        if not is_school_leader(request.user):
            raise AppError("FORBIDDEN", "Only school leadership can dispatch notifications.", 403)

        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        validated = serializer.validated_data
        title = str(validated.get("title") or "").strip()
        message = str(validated.get("message") or "").strip()
        category = str(validated.get("category") or "ANNOUNCEMENT").strip().upper()
        student = validated.get("student")
        explicit_recipient = validated.get("recipient")

        if explicit_recipient:
            notif = send_user_notification(
                recipient=explicit_recipient,
                title=title,
                message=message,
                category=category,
                student=student,
            )
            data = self.get_serializer(notif).data if notif else {"ok": True}
            return Response(data, status=status.HTTP_201_CREATED)

        # Broadcast from admin console: deliver to target role (default: active parents + staff)
        target_role = str(request.data.get("target_role") or "ALL").strip().upper()
        recipients_qs = User.objects.filter(is_active=True).select_related("profile")
        if target_role in ("PARENT", "FACULTY", "DEPT_HEAD", "PRINCIPAL", "ADMIN"):
            recipients_qs = recipients_qs.filter(profile__role=target_role)
        recipients = list(recipients_qs[:1000])

        first_notif = None
        sent_count = 0
        for rec in recipients:
            notif = send_user_notification(
                recipient=rec,
                title=title,
                message=message,
                category=category,
                student=student,
            )
            if notif:
                sent_count += 1
                if first_notif is None:
                    first_notif = notif

        payload = self.get_serializer(first_notif).data if first_notif else {}
        payload["ok"] = True
        payload["dispatched_count"] = sent_count
        return Response(payload, status=status.HTTP_201_CREATED)

    def update(self, request, *args, **kwargs):
        instance = self.get_object()
        if not is_school_leader(request.user):
            if set(request.data.keys()) - {"is_read"}:
                raise AppError("FORBIDDEN", "You can only update read status on your notifications.", 403)
            instance.is_read = bool(request.data.get("is_read", True))
            instance.save(update_fields=["is_read"])
            return Response(self.get_serializer(instance).data)
        return super().update(request, *args, **kwargs)

    def partial_update(self, request, *args, **kwargs):
        instance = self.get_object()
        if not is_school_leader(request.user):
            if set(request.data.keys()) - {"is_read"}:
                raise AppError("FORBIDDEN", "You can only update read status on your notifications.", 403)
            instance.is_read = bool(request.data.get("is_read", True))
            instance.save(update_fields=["is_read"])
            return Response(self.get_serializer(instance).data)
        return super().partial_update(request, *args, **kwargs)

    def destroy(self, request, *args, **kwargs):
        if not is_school_leader(request.user):
            raise AppError("FORBIDDEN", "Only school leadership can delete notifications.", 403)
        return super().destroy(request, *args, **kwargs)

class ChatMessageView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    def get(self, request):
        student_id = request.query_params.get("student_id")
        teacher_id = request.query_params.get("teacher_id") or request.query_params.get("faculty_id")
        if not student_id or not str(student_id).isdigit():
            return Response({"error": "Valid student_id is required."}, status=status.HTTP_400_BAD_REQUEST)
        t_id = int(teacher_id) if (teacher_id and str(teacher_id).isdigit()) else None
        return Response(get_thread_messages(user=request.user, student_id=int(student_id), teacher_id=t_id))

    def post(self, request):
        student_id = request.data.get("student_id")
        teacher_id = request.data.get("teacher_id") or request.data.get("faculty_id")
        if not student_id or not str(student_id).isdigit():
            return Response({"error": "Valid student_id is required."}, status=status.HTTP_400_BAD_REQUEST)
        t_id = int(teacher_id) if (teacher_id and str(teacher_id).isdigit()) else None

        student, teacher, parent_user = authorize_chat_thread(
            user=request.user, student_id=int(student_id), teacher_id=t_id
        )
        msg = create_chat_message(
            sender=request.user,
            student=student,
            teacher=teacher,
            parent_user=parent_user,
            body=request.data.get("body") or request.data.get("message") or "",
        )
        return Response(
            {
                "id": msg.id,
                "sender_id": msg.sender_id,
                "sender_name": request.user.get_full_name() or request.user.username,
                "is_mine": True,
                "body": msg.body,
                "is_read": False,
                "created_at": timezone.localtime(msg.created_at).strftime("%Y-%m-%d %I:%M %p"),
                "time_only": timezone.localtime(msg.created_at).strftime("%I:%M %p"),
            },
            status=status.HTTP_201_CREATED,
        )

    def put(self, request):
        msg_id = request.data.get("id") or request.data.get("message_id")
        if not msg_id or not str(msg_id).isdigit():
            raise AppError("INVALID_ID", "Valid numeric message id is required.", 400)
        msg = update_chat_message(
            user=request.user,
            message_id=int(msg_id),
            new_body=request.data.get("body") or request.data.get("message") or "",
        )
        return Response({"ok": True, "id": msg.id, "body": msg.body})

    def patch(self, request):
        return self.put(request)

    def delete(self, request):
        msg_id = request.data.get("id") or request.query_params.get("id")
        if not msg_id or not str(msg_id).isdigit():
            raise AppError("INVALID_ID", "Valid numeric message id is required.", 400)
        delete_chat_message(user=request.user, message_id=int(msg_id))
        return Response({"ok": True, "message": "Message deleted."})

def _materialize_dashboard_photos(payload: dict, request=None) -> dict:
    out = copy.deepcopy(payload)
    for item in out.get("recent_scans", []):
        raw_path = item.pop("_photo_path", None)
        item["photo_url"] = build_media_url(f"/media/{raw_path}", request) if raw_path else None
    return out


def get_dashboard_overview(*, user, request=None) -> dict:
    started = time_module.perf_counter()
    leader = is_school_leader(user)
    cache_key = "dashboard:leader" if leader else f"dashboard:teacher:{user.id}"
    cached = cache.get(cache_key)
    if cached:
        out = _materialize_dashboard_photos(cached, request)
        out["analytics"]["average_latency_ms"] = max(1, round((time_module.perf_counter() - started) * 1000))
        return out

    today = timezone.localdate()
    today_start = timezone.make_aware(datetime.combine(today, datetime.min.time()))

    if leader:
        total_students = Student.objects.filter(is_active=True).count()
        present_ids = set(
            DailyAttendance.objects.filter(
                attendance_date=today,
                student__is_active=True,
                status__in=["PRESENT", "LATE"],
            ).values_list("student_id", flat=True)
        )
        cutting_count = DailyAttendance.objects.filter(
            attendance_date=today, student__is_active=True, status="CUTTING"
        ).count()
        total_teachers = TeacherProfile.objects.filter(is_active=True).count()
        present_teachers = (
            TeacherLocationLog.objects.filter(recorded_at__gte=today_start, is_inside_geofence=True)
            .values("teacher_id")
            .distinct()
            .count()
        )
        recent_qs = ClassAttendance.objects.filter(attendance_date=today)
        alerts_today = Notification.objects.filter(created_at__gte=today_start).count()
        linked_parents = UserProfile.objects.filter(role="PARENT", user__is_active=True).count()
    else:
        teacher = resolve_teacher_for_user(user)
        student_ids = get_teacher_student_ids(teacher)
        total_students = len(student_ids)
        present_ids = set(
            DailyAttendance.objects.filter(
                attendance_date=today, student_id__in=student_ids, status__in=["PRESENT", "LATE"]
            ).values_list("student_id", flat=True)
        )
        cutting_count = DailyAttendance.objects.filter(
            attendance_date=today, student_id__in=student_ids, status="CUTTING"
        ).count()
        total_teachers = 1 if teacher else 0
        present_teachers = int(
            bool(
                teacher
                and TeacherLocationLog.objects.filter(
                    teacher=teacher, recorded_at__gte=today_start, is_inside_geofence=True
                ).exists()
            )
        )
        recent_qs = ClassAttendance.objects.filter(attendance_date=today, student_id__in=student_ids)
        alerts_today = Notification.objects.filter(student_id__in=student_ids, created_at__gte=today_start).count()
        linked_parents = (
            Student.objects.filter(id__in=student_ids, parent_user__isnull=False)
            .values("parent_user_id")
            .distinct()
            .count()
        )

    hour_rows = (
        recent_qs.annotate(hr=ExtractHour("scanned_at"))
        .values("hr")
        .annotate(cnt=Count("id"))
    )
    hour_map = {r["hr"]: r["cnt"] for r in hour_rows}
    peak = max(hour_map.values(), default=1) or 1
    hourly_scans = [
        {
            "hour": f"{h % 12 or 12} {'AM' if h < 12 else 'PM'}",
            "count": hour_map.get(h, 0),
            "height": round((hour_map.get(h, 0) / peak) * 100),
            "is_peak": hour_map.get(h, 0) == peak and hour_map.get(h, 0) > 0,
        }
        for h in range(6, 18)
    ]

    recent_logs = list(
        recent_qs.select_related(
            "student",
            "class_period__subject",
            "class_period__section",
            "class_period__section__grade_level",
        ).order_by("-scanned_at")[:6]
    )
    recent_scans = [
        {
            "id": item.id,
            "person_name": item.student.full_name,
            "_photo_path": item.student.photo.name if (item.student.photo and hasattr(item.student.photo, "name")) else None,
            "status": item.status,
            "subject_code": item.class_period.subject.code if (item.class_period and item.class_period.subject) else "CLASS",
            "grade_section": format_section_label(

                else "",
                item.class_period.section.name if (item.class_period and item.class_period.section) else "",
            ),
            "scan_time": timezone.localtime(item.scanned_at).strftime("%I:%M:%S %p"),
        }
        for item in recent_logs
    ]

    present_count = len(present_ids)
    payload = {
        "total_students": total_students,
        "students_present": present_count,
        "students_on_campus": present_count,
        "students_absent": max(0, total_students - present_count),
        "students_cutting": cutting_count,
        "attendance_rate": round((present_count / total_students) * 100, 1) if total_students else 0.0,

def get_dropout_risk_report(*, user, section_id: int | None = None, request=None) -> dict:
    today = timezone.localdate()
    window_start = today - timedelta(days=30)

    qs = Enrollment.objects.filter(
        status__in=["ENROLLED", "TRANSFERRED_IN"], student__is_active=True
    ).select_related("student", "section", "section__grade_level")
    active_sy = SchoolYear.objects.filter(is_active=True).first()
    if active_sy:
        qs = qs.filter(school_year=active_sy)
    if not is_school_leader(user):
        teacher = resolve_teacher_for_user(user)
        if not teacher:
            return {
                "window_days": 30,
                "total_evaluated": 0,
                "critical_sardo_count": 0,
                "early_warning_count": 0,
                "at_risk_students": [],
            }
        qs = qs.filter(section_id__in=get_teacher_section_ids(teacher))
    if section_id:
        qs = qs.filter(section_id=section_id)

    enrollments = list(qs)
    student_ids = [e.student_id for e in enrollments]

    stats = {
        r["student_id"]: r
        for r in DailyAttendance.objects.filter(student_id__in=student_ids, attendance_date__gte=window_start)
        .values("student_id")
        .annotate(
            absences=Count("id", filter=Q(status="ABSENT")),
            cutting=Count("id", filter=Q(status="CUTTING")),
            lates=Count("id", filter=Q(status="LATE")),
        )
    }
    morning_lates = dict(
        ClassAttendance.objects.filter(
            student_id__in=student_ids,
            attendance_date__gte=window_start,
            status="LATE",
            class_period__start_time__lte=time(8, 30),
        )
        .values("student_id")
        .annotate(cnt=Count("id"))
        .values_list("student_id", "cnt")
    )

    at_risk = []
    critical = warning = 0
    for enr in enrollments:
        st = enr.student
        row = stats.get(st.id, {})
        absences, cutting, lates = row.get("absences", 0), row.get("cutting", 0), row.get("lates", 0)
        m_lates = morning_lates.get(st.id, 0)

        score = min(100, (absences * 15) + (cutting * 20) + (lates * 5) + (m_lates * 3))
        if score < 20:
            continue

        is_crit = absences >= 5 or cutting >= 3 or score >= 60
        if is_crit:
            critical += 1
        else:
            warning += 1

        at_risk.append({
            "student_id": st.id,
            "lrn": st.lrn,
            "full_name": st.full_name,
            "section": format_section_label(
                enr.section.grade_level.name if (enr.section and enr.section.grade_level) else "",

def get_monthly_dtr_data(
    *,
    user,
    month: int,
    year: int,
    teacher_id: int | None = None,
    list_only: bool = False,
    request=None,
) -> dict:
    """Computes CSC Form 48 rows strictly from real database GPS logs and boundary alerts."""
    from common.permissions import get_user_role
    from apps.campus.models import BoundaryAlert

    month = max(1, min(12, int(month)))
    year = int(year)
    if not (1900 <= year <= 2100):
        raise AppError("INVALID_YEAR", "Year must be between 1900 and 2100.", 400)
    role = get_user_role(user)
    leader = is_school_leader(user)
    own_teacher = resolve_teacher_for_user(user)

    month_start_dt = timezone.make_aware(datetime(year, month, 1, 0, 0, 0))
    next_month_start_dt = timezone.make_aware(
        datetime(year + 1, 1, 1, 0, 0, 0) if month == 12 else datetime(year, month + 1, 1, 0, 0, 0)
    )

    if list_only and not teacher_id:
        teachers = (
            list(TeacherProfile.objects.filter(is_active=True).order_by("last_name", "first_name"))
            if leader
            else ([own_teacher] if own_teacher else [])
        )
        t_ids = [t.id for t in teachers]
        present_days_by_teacher: dict[int, set[int]] = {}
        for tid, rec_at in TeacherLocationLog.objects.filter(
            teacher_id__in=t_ids,
            recorded_at__gte=month_start_dt,
            recorded_at__lt=next_month_start_dt,
            is_inside_geofence=True,
        ).values_list("teacher_id", "recorded_at"):
            present_days_by_teacher.setdefault(tid, set()).add(timezone.localtime(rec_at).day)
        present_counts = {tid: len(days) for tid, days in present_days_by_teacher.items()}
        existing_dtrs = {
            d.teacher_id: d
            for d in MonthlyTimeRecord.objects.filter(teacher_id__in=t_ids, month=month, year=year)
        }
        return {
            "month": calendar.month_name[month],
            "month_number": month,
            "year": year,
            "records": [
                {
                    "id": t.id,
                    "dtr_id": existing_dtrs[t.id].id if t.id in existing_dtrs else None,
                    "faculty_id": t.id,
                    "employee_id": t.employee_id,
                    "faculty_name": t.full_name,
                    "position": t.position,
                    "department": t.department,
                    "month": month,
                    "year": year,
                    "days_present": present_counts.get(t.id, 0),
                    "is_submitted": bool(t.id in existing_dtrs and existing_dtrs[t.id].is_submitted),
                    "is_dept_head_approved": bool(

        if not leader and (not own_teacher or own_teacher.id != teacher_id):
            raise AppError("FORBIDDEN", "You can only view your own Daily Time Record.", 403)
        teacher = TeacherProfile.objects.filter(id=teacher_id).first()
    else:
        teacher = own_teacher or (
            TeacherProfile.objects.filter(is_active=True).order_by("last_name", "first_name").first()
            if leader
            else None
        )

    if not teacher:
        raise AppError("TEACHER_NOT_FOUND", "No teacher profile is linked to this account.", 404)

    dtr, _ = MonthlyTimeRecord.objects.get_or_create(teacher=teacher, month=month, year=year)
    _, num_days = calendar.monthrange(year, month)

    pings_by_day: dict[int, list] = {}
    for log in TeacherLocationLog.objects.filter(
        teacher=teacher,

    alerts_by_day: dict[int, BoundaryAlert] = {}

                "is_loafing": False,
                "loafing_excused": False,
                "loafing_remarks": "",
            })
            continue

        dt_obj = date(year, month, day_num)
        is_weekend = dt_obj.weekday() >= 5
        day_pings = pings_by_day.get(day_num, [])
        am_pings = [p for p in day_pings if p.hour < 12]
        pm_pings = [p for p in day_pings if p.hour >= 12]

        if day_pings:
            days_present += 1

        alert = alerts_by_day.get(day_num)
        has_alert = alert is not None
        is_excused = bool(alert and str(alert.status).upper() == "EXCUSED")

        rows.append({

            "is_weekend": is_weekend,

            "am_departure": am_pings[-1].strftime("%I:%M") if len(am_pings) > 1 else "",
            "pm_arrival": pm_pings[0].strftime("%I:%M") if pm_pings else "",

            "status": "PRESENT" if day_pings else ("WEEKEND" if is_weekend else "NO_LOG"),
            "is_loafing": has_alert,
            "loafing_excused": is_excused,

        })

    school_meta = get_school_metadata(request)

    return {

        "faculty_id": teacher.id,
        "teacher_id": teacher.id,

        "saturdays_hours": dtr.saturday_hours or "",
        "regular_hours": dtr.regular_hours or "",
        "saturday_hours": dtr.saturday_hours or "",
        "school_head": school_meta.get("school_head", ""),
        "school_head_title": school_meta.get("principal_title", ""),
        "dtr_submitted": bool(dtr.is_submitted),
        "is_submitted": bool(dtr.is_submitted),

from django.db import transaction

def _clean_text(val: object, max_len: int = 120) -> str:
    return str(val or "").replace("<", "").replace(">", "").strip()[:max_len]


@transaction.atomic
def process_monthly_dtr_action(
    *,
    request,
    action: str,
    month: int,
    year: int,
    teacher_id: int | None = None,
    data: dict | None = None,
) -> dict:
    payload = data or {}
    user = request.user
    role = get_user_role(user)
    leader = is_school_leader(user)
    own_teacher = resolve_teacher_for_user(user)

    target_id = teacher_id or payload.get("faculty_id") or payload.get("teacher_id")
    teacher = (
        TeacherProfile.objects.filter(id=int(target_id)).first()
        if (target_id and str(target_id).isdigit())
        else own_teacher
    )
    if not teacher:
        raise AppError("TEACHER_NOT_FOUND", "Teacher profile not found.", 404)

    clean_action = str(action or "").strip().upper()

    if clean_action == "SUBMIT_DTR":
        if not leader and (not own_teacher or own_teacher.id != teacher.id):
            raise AppError("FORBIDDEN", "You can only submit your own Daily Time Record.", 403)

        dtr, _ = MonthlyTimeRecord.objects.get_or_create(teacher=teacher, month=month, year=year)
        if dtr.is_dept_head_approved and not leader:
            raise AppError("ALREADY_APPROVED", "Approved DTRs are locked against edits.", 400)

        reg = _clean_text(payload.get("regular_days_hours") or payload.get("regular_hours"), 100)
        sat = _clean_text(payload.get("saturdays_hours") or payload.get("saturday_hours"), 100)
        if reg:
            dtr.regular_hours = reg
        if sat:
            dtr.saturday_hours = sat
        dtr.is_submitted = True
        if hasattr(dtr, "submitted_at"):
            dtr.submitted_at = timezone.now()
        dtr.save()

        record_audit_log(
            request=request,
            table_name=dtr._meta.db_table,
            record_id=dtr.id,
            action="SUBMIT",
            new_data={"month": month, "year": year, "teacher_id": teacher.id},
        )
        return {"ok": True, "message": "Daily Time Record certified and submitted."}

    if clean_action == "APPROVE_DEPT_HEAD":
        if not leader:
            raise AppError("FORBIDDEN", "Only Department Heads, Principals, or Admins can approve DTRs.", 403)

        dtr, _ = MonthlyTimeRecord.objects.get_or_create(teacher=teacher, month=month, year=year)
        if "regular_days_hours" in payload or "regular_hours" in payload:
            reg = _clean_text(payload.get("regular_days_hours") or payload.get("regular_hours"), 100)
            if reg:
                dtr.regular_hours = reg
        if "saturdays_hours" in payload or "saturday_hours" in payload:
            sat = _clean_text(payload.get("saturdays_hours") or payload.get("saturday_hours"), 100)
            if sat:
                dtr.saturday_hours = sat

        dtr.is_submitted = True
        dtr.is_dept_head_approved = True
        if hasattr(dtr, "approved_by"):
            dtr.approved_by = user
        if hasattr(dtr, "dept_head_approved_at"):
            dtr.dept_head_approved_at = timezone.now()
        dtr.save()

        record_audit_log(
            request=request,
            table_name=dtr._meta.db_table,
            record_id=dtr.id,
            action="APPROVE",
            new_data={"month": month, "year": year, "teacher_id": teacher.id},
        )
        return {"ok": True, "message": f"Official DTR approved for {teacher.full_name}."}

    if clean_action == "EXCUSE_LOAFING":
        if not (user.is_superuser or role in ("ADMIN", "SUPERADMIN", "PRINCIPAL")):
            raise AppError("FORBIDDEN", "Only the School Principal or System Admin can excuse campus alerts.", 403)

        raw_date = str(payload.get("date_str") or "").strip()
        remarks = _clean_text(payload.get("remarks"), 240)
        if not remarks:
            raise AppError("REMARKS_REQUIRED", "Official remarks are required to excuse an alert.", 400)

        incident_dt = parse_date(raw_date)
        if not incident_dt:
            raise AppError("INVALID_DATE", "Valid incident date (YYYY-MM-DD) is required.", 400)

        alerts_qs = BoundaryAlert.objects.filter(teacher=teacher, incident_date=incident_dt)
        updated_count = alerts_qs.update(status="EXCUSED", remarks=remarks)
        if updated_count == 0:
            raise AppError("ALERT_NOT_FOUND", f"No boundary alert found for {teacher.full_name} on {incident_dt}.", 404)

        record_audit_log(
            request=request,
            table_name=BoundaryAlert._meta.db_table,
            record_id=teacher.id,
            action="EXCUSE",
            new_data={"date": str(incident_dt), "remarks": remarks, "updated": updated_count},
        )
        return {"ok": True, "message": f"Boundary alert on {incident_dt} marked as excused."}

    raise AppError("INVALID_ACTION", f"Unsupported DTR action: {clean_action}", 400)

def compute_age(birthdate: date | None, cutoff: date) -> int | str:
    if not birthdate:
        return ""
    return max(0, cutoff.year - birthdate.year - ((cutoff.month, cutoff.day) < (birthdate.month, birthdate.day)))


def generate_sf1_report(*, section_id: int, school_year: str | None = None, request=None) -> dict:

    sy = section.school_year
    sy_code = str(school_year or (sy.code if sy else "2026-2027"))
    base_yr = int(sy_code.split("-")[0]) if "-" in sy_code and sy_code.split("-")[0].isdigit() else 2026
    june_1 = date(base_yr, 6, 1)
    cutoff = (sy.baseline_cutoff_date if sy else None) or (june_1 + timedelta(days=(4 - june_1.weekday()) % 7))

    enrollments = (
        Enrollment.objects.filter(
            section=section,

    )

    males, females, combined = [], [], []
    for enr in enrollments:
        st = enr.student
        is_male = str(st.sex).upper().startswith("M")
        address = ", ".join(filter(None, [st.house_street_sitio, st.barangay, st.municipality_city, st.province]))

        "total_female": len(females),
        "total_combined": len(combined),
    }

MONTH_NAMES = [
    "January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December",
]
MONTH_ABBR = {name[:3].lower(): idx + 1 for idx, name in enumerate(MONTH_NAMES)}


def resolve_month(raw_month) -> tuple[int, str]:
    clean = str(raw_month or "").strip()
    if not clean:
        return 10, "October"
    if clean.isdigit():
        num = int(clean)
        if 1 <= num <= 12:
            return num, MONTH_NAMES[num - 1]
        raise AppError("INVALID_MONTH", "Month must be between 1 and 12.", 400)
    cap = clean.capitalize()
    if cap in MONTH_NAMES:
        return MONTH_NAMES.index(cap) + 1, cap
    abbr_num = MONTH_ABBR.get(clean[:3].lower())
    if abbr_num:
        return abbr_num, MONTH_NAMES[abbr_num - 1]
    raise AppError("INVALID_MONTH", f"Invalid month '{raw_month}'.", 400)


def _as_date(val) -> date | None:
    if isinstance(val, datetime):
        return val.date()
    return val


def make_metric(m=0, f=0) -> dict:
    return {"m": m, "f": f, "total": m + f}


def has_five_consecutive_absences(att_map: dict, school_day_nums: list[int]) -> bool:
    streak = 0
    for d in school_day_nums:
        if att_map.get(d, "") in ("absent", "x", "a"):
            streak += 1
            if streak >= 5:
                return True
        else:
            streak = 0
    return False


def generate_sf2_report(*, section_id: int, month_name: str = "October", year: int = 2026, request=None) -> dict:
    section = Section.objects.select_related("grade_level", "school_year", "adviser").filter(id=section_id).first()
    if not section:
        raise AppError("SECTION_NOT_FOUND", f"Section {section_id} not found.", 404)

    year = int(year)
    if not (1900 <= year <= 2100):
        raise AppError("INVALID_YEAR", "Year must be between 1900 and 2100.", 400)
    month_num, clean_month = resolve_month(month_name)
    _, num_days = calendar.monthrange(year, month_num)
    month_start, month_end = date(year, month_num, 1), date(year, month_num, num_days)

    holiday_dates = set(
        SchoolHoliday.objects.filter(holiday_date__range=(month_start, month_end)).values_list("holiday_date", flat=True)
    )
    abbr = {0: "M", 1: "T", 2: "W", 3: "TH", 4: "F"}
    school_days, day_nums = [], []
    for d in range(1, num_days + 1):
        dt = date(year, month_num, d)
        if dt.weekday() < 5 and dt not in holiday_dates:
            day_nums.append(d)
            school_days.append({"dateNumber": d, "dayOfWeek": abbr[dt.weekday()], "fullDate": str(dt)})
    school_day_set = set(day_nums)

    enrollments = list(
        Enrollment.objects.filter(section=section, student__is_active=True)
        .select_related("student")
        .order_by("student__last_name", "student__first_name")
    )
    student_ids = [e.student_id for e in enrollments]

    records_by_st: dict[int, dict[int, str]] = {}
    for row in DailyAttendance.objects.filter(
        section=section, student_id__in=student_ids, attendance_date__range=(month_start, month_end)
    ).values("student_id", "attendance_date", "status"):
        day_val = row["attendance_date"].day
        if day_val in school_day_set:
            records_by_st.setdefault(row["student_id"], {})[day_val] = row["status"].lower()

    for row in (
        ClassAttendance.objects.filter(
            class_period__section=section,
            student_id__in=student_ids,
            attendance_date__range=(month_start, month_end),
        )
        .order_by("scanned_at")
        .values("student_id", "attendance_date", "status")
    ):
        day_val = row["attendance_date"].day
        if day_val in school_day_set:
            records_by_st.setdefault(row["student_id"], {}).setdefault(day_val, row["status"].lower())

    sy = section.school_year
    sy_start_year = year
    if sy and sy.code and "-" in sy.code and sy.code.split("-")[0].isdigit():
        sy_start_year = int(sy.code.split("-")[0])
    elif sy and sy.start_date:
        sy_start_year = sy.start_date.year
    june_1 = date(sy_start_year, 6, 1)
    cutoff = (
        sy.baseline_cutoff_date
        if (sy and sy.baseline_cutoff_date)
        else (june_1 + timedelta(days=(4 - june_1.weekday()) % 7))
    )

    males, females = [], []
    reg_m = reg_f = 0
    consec_m = consec_f = 0
    june_m = june_f = late_m = late_f = 0
    drop_m = drop_f = tout_m = tout_f = tin_m = tin_f = 0

    for enr in enrollments:
        st = enr.student
        is_m = str(st.sex or "").upper().startswith("M")
        enr_date = _as_date(enr.enrollment_date) or month_start
        exit_dt = _as_date(enr.status_date)
        st_code = str(enr.status or "").upper()

        # Skip learners enrolled after the report month or who exited prior to this month
        if enr_date > month_end:
            continue
        if st_code in ("DROPPED", "TRANSFERRED_OUT") and exit_dt and exit_dt < month_start:
            continue

        att = {d: v for d, v in records_by_st.get(st.id, {}).items() if d in school_day_set}
        rec = {
            "id": st.id,
            "lrn": st.lrn,
            "name": st.full_name,
            "sex": "M" if is_m else "F",
            "attendance": att,
            "total_absent": sum(1 for d in day_nums if att.get(d, "") in ("absent", "x", "a")),
            "total_tardy": sum(1 for d in day_nums if att.get(d, "") in ("late", "tardy", "t")),
            "remarks": enr.remarks,
        }
        if is_m:
            males.append(rec)
            consec_m += int(has_five_consecutive_absences(att, day_nums))
        else:
            females.append(rec)
            consec_f += int(has_five_consecutive_absences(att, day_nums))

        if enr_date <= cutoff:
            if is_m:
                june_m += 1
            else:
                june_f += 1
        else:
            if is_m:
                late_m += 1
            else:
                late_f += 1

        exited_this_month = False
        if st_code == "DROPPED" and (not exit_dt or (month_start <= exit_dt <= month_end)):
            exited_this_month = True
            if is_m:
                drop_m += 1
            else:
                drop_f += 1
        elif st_code == "TRANSFERRED_OUT" and (not exit_dt or (month_start <= exit_dt <= month_end)):
            exited_this_month = True
            if is_m:
                tout_m += 1
            else:
                tout_f += 1
        elif st_code == "TRANSFERRED_IN" and (month_start <= enr_date <= month_end):
            if is_m:
                tin_m += 1
            else:
                tin_f += 1

        if not exited_this_month:
            if is_m:
                reg_m += 1
            else:
                reg_f += 1

    n_days = len(school_days) or 1
    present_vals = {"present", "late", "tardy", "cutting", "excused"}
    ada_m = round(
        sum(1 for m in males for d in day_nums if m["attendance"].get(d, "") in present_vals) / n_days,
        1,
    )
    ada_f = round(
        sum(1 for f in females for d in day_nums if f["attendance"].get(d, "") in present_vals) / n_days,
        1,
    )
    ada_tot = round(ada_m + ada_f, 1)

    reg_all = reg_m + reg_f
    tot_june = june_m + june_f

    return {
        **get_school_metadata(request),
        "academic_year": section.school_year.code if section.school_year else "",
        "grade_level": section.grade_level.name if section.grade_level else "",
        "section_name": section.name,
        "month": clean_month,
        "year": year,
        "adviser_name": section.adviser.full_name if section.adviser else "",
        "school_days": school_days,
        "has_enrolled_students": len(males) + len(females) > 0,
        "males": males,
        "females": females,
        "metrics": {
            "enrolment_june": make_metric(june_m, june_f),
            "late_enrolment": make_metric(late_m, late_f),
            "registered_end": make_metric(reg_m, reg_f),
            "percentage_enrolment": {
                "m": round((reg_m / june_m) * 100, 1) if june_m else 0.0,
                "f": round((reg_f / june_f) * 100, 1) if june_f else 0.0,
                "total": round((reg_all / tot_june) * 100, 1) if tot_june else 0.0,
            },
            "average_daily_attendance": {"m": ada_m, "f": ada_f, "total": ada_tot},
            "percentage_attendance": {
                "m": round((ada_m / reg_m) * 100, 1) if reg_m else 0.0,
                "f": round((ada_f / reg_f) * 100, 1) if reg_f else 0.0,
                "total": round((ada_tot / reg_all) * 100, 1) if reg_all else 0.0,
            },
            "consecutive_5_absent_count": make_metric(consec_m, consec_f),
            "consecutive_5": make_metric(consec_m, consec_f),
            "drop_out": make_metric(drop_m, drop_f),
            "transferred_out": make_metric(tout_m, tout_f),
            "transferred_in": make_metric(tin_m, tin_f),
        },
    }

    "January": 1, "February": 2, "March": 3, "April": 4, "May": 5, "June": 6,

def make_triplet(m=0, f=0) -> dict:
    total = m + f
    return {
        "m": round(m, 1) if isinstance(m, float) else int(m),
        "f": round(f, 1) if isinstance(f, float) else int(f),
        "total": round(total, 1) if isinstance(total, float) else int(total),
    }


def add_triplets(t1: dict, t2: dict) -> dict:
    return make_triplet(t1["m"] + t2["m"], t1["f"] + t2["f"])


def compute_attendance_pct(ada: dict, reg: dict) -> dict:
    return {
        "m": round((ada["m"] / reg["m"]) * 100, 1) if reg["m"] > 0 else 0.0,
        "f": round((ada["f"] / reg["f"]) * 100, 1) if reg["f"] > 0 else 0.0,
        "total": round((ada["total"] / reg["total"]) * 100, 1) if reg["total"] > 0 else 0.0,
    }


def empty_summary_row(section_id: str, section_name: str, grade_level: str) -> dict:
    zero = make_triplet(0, 0)
    return {
        "section_id": section_id,
        "section_name": section_name,
        "grade_level": grade_level,
        "adviser_name": "",
        "registered_end": dict(zero),
        "daily_average": make_triplet(0.0, 0.0),
        "attendance_percentage": {"m": 0.0, "f": 0.0, "total": 0.0},
        "dropped_out_prev": dict(zero),
        "dropped_out_month": dict(zero),
        "dropped_out_cumulative": dict(zero),
        "transferred_out_prev": dict(zero),
        "transferred_out_month": dict(zero),
        "transferred_out_cumulative": dict(zero),
        "transferred_in_prev": dict(zero),
        "transferred_in_month": dict(zero),
        "transferred_in_cumulative": dict(zero),
    }


SUMMABLE_KEYS = (

    "daily_average",
    "dropped_out_prev",
    "dropped_out_month",
    "dropped_out_cumulative",
    "transferred_out_prev",

def generate_sf4_report(*, month_name: str = "October", year: int = 2026, school_year: str = "", request=None) -> dict:
    month_num, clean_month = resolve_month(month_name)
    year = int(year)
    if not (1900 <= year <= 2100):
        raise AppError("INVALID_YEAR", "Year must be between 1900 and 2100.", 400)
    _, num_days = calendar.monthrange(year, month_num)
    start_dt, end_dt = date(year, month_num, 1), date(year, month_num, num_days)

    holiday_dates = set(
        SchoolHoliday.objects.filter(holiday_date__range=(start_dt, end_dt)).values_list("holiday_date", flat=True)
    )
    school_days = [
        date(year, month_num, d)
        for d in range(1, num_days + 1)
        if date(year, month_num, d).weekday() < 5 and date(year, month_num, d) not in holiday_dates
    ]
    n_days = len(school_days) or 1

    target_sy = None
    if school_year:
        sy_clean = str(school_year).strip()
        target_sy = (
            SchoolYear.objects.filter(id=int(sy_clean)).first()
            if sy_clean.isdigit()
            else SchoolYear.objects.filter(code__iexact=sy_clean).first()
        )
    if target_sy is None:
        target_sy = SchoolYear.objects.filter(is_active=True).first() or SchoolYear.objects.first()

    sections_qs = Section.objects.select_related("grade_level", "school_year", "adviser").filter(is_active=True)
    if target_sy:
        sections_qs = sections_qs.filter(Q(school_year=target_sy) | Q(school_year__isnull=True))
    sections = list(sections_qs.order_by("grade_level__level_order", "name"))

    enr_qs = Enrollment.objects.filter(section__in=sections, student__is_active=True).select_related("student")
    if target_sy:
        enr_qs = enr_qs.filter(Q(school_year=target_sy) | Q(school_year__isnull=True))

    enr_by_sec: dict[int, list] = {}
    all_student_ids = set()
    for enr in enr_qs:
        enr_by_sec.setdefault(enr.section_id, []).append(enr)
        all_student_ids.add(enr.student_id)

    present_codes = ["PRESENT", "LATE", "CUTTING", "EXCUSED"]
    present_pairs = set()
    if all_student_ids:
        present_pairs.update(
            DailyAttendance.objects.filter(
                section__in=sections,
                student_id__in=all_student_ids,
                attendance_date__range=(start_dt, end_dt),
                status__in=present_codes,
            ).values_list("student_id", "attendance_date")
        )
        present_pairs.update(
            ClassAttendance.objects.filter(
                class_period__section__in=sections,
                student_id__in=all_student_ids,
                attendance_date__range=(start_dt, end_dt),
                status__in=present_codes,
            ).values_list("student_id", "attendance_date")
        )

    grade_names = list(GradeLevel.objects.order_by("level_order").values_list("name", flat=True))
    groups_map: dict[str, list] = {g: [] for g in grade_names if g}

    for sec in sections:
        g_label = (sec.grade_level.name if sec.grade_level else "NON-GRADED").strip() or "NON-GRADED"
        adviser_label = (sec.adviser.full_name if sec.adviser else "Unassigned").upper()

        reg_m = reg_f = att_m = att_f = 0
        drop_prev_m = drop_prev_f = drop_mo_m = drop_mo_f = 0
        tout_prev_m = tout_prev_f = tout_mo_m = tout_mo_f = 0
        tin_prev_m = tin_prev_f = tin_mo_m = tin_mo_f = 0

        for enr in enr_by_sec.get(sec.id, []):
            st = enr.student
            is_m = str(st.sex or "").upper().startswith("M")
            enr_dt = enr.enrollment_date.date() if isinstance(enr.enrollment_date, datetime) else enr.enrollment_date
            if enr_dt and enr_dt > end_dt:
                continue

            status_val = str(enr.status or "").upper()
            exit_dt = enr.status_date.date() if isinstance(enr.status_date, datetime) else enr.status_date

            if "DROP" in status_val:
                if exit_dt and exit_dt < start_dt:
                    if is_m:
                        drop_prev_m += 1
                    else:
                        drop_prev_f += 1
                    continue
                if not exit_dt or exit_dt <= end_dt:
                    if is_m:
                        drop_mo_m += 1
                    else:
                        drop_mo_f += 1
            elif "TRANSFERRED" in status_val and "OUT" in status_val:
                if exit_dt and exit_dt < start_dt:
                    if is_m:
                        tout_prev_m += 1
                    else:
                        tout_prev_f += 1
                    continue
                if not exit_dt or exit_dt <= end_dt:
                    if is_m:
                        tout_mo_m += 1
                    else:
                        tout_mo_f += 1

            if "TRANSFERRED" in status_val and "IN" in status_val:
                if enr_dt and enr_dt < start_dt:
                    if is_m:
                        tin_prev_m += 1
                    else:
                        tin_prev_f += 1
                else:
                    if is_m:
                        tin_mo_m += 1
                    else:
                        tin_mo_f += 1

            if status_val not in ("DROPPED", "TRANSFERRED_OUT") or (exit_dt and exit_dt > end_dt):
                if is_m:
                    reg_m += 1
                else:
                    reg_f += 1

            for s_day in school_days:
                if (st.id, s_day) in present_pairs:
                    if is_m:
                        att_m += 1
                    else:
                        att_f += 1

        reg = make_triplet(reg_m, reg_f)
        ada = make_triplet(round(att_m / n_days, 1), round(att_f / n_days, 1))
        d_prev, d_mo = make_triplet(drop_prev_m, drop_prev_f), make_triplet(drop_mo_m, drop_mo_f)
        to_prev, to_mo = make_triplet(tout_prev_m, tout_prev_f), make_triplet(tout_mo_m, tout_mo_f)
        ti_prev, ti_mo = make_triplet(tin_prev_m, tin_prev_f), make_triplet(tin_mo_mo if False else tin_mo_m, tin_mo_f)

        groups_map.setdefault(g_label, []).append({
            "section_id": sec.id,
            "section_name": sec.name,
            "grade_level": g_label,
            "adviser_name": adviser_label,
            "registered_end": reg,
            "daily_average": ada,
            "attendance_percentage": compute_attendance_pct(ada, reg),
            "dropped_out_prev": d_prev,
            "dropped_out_month": d_mo,
            "dropped_out_cumulative": add_triplets(d_prev, d_mo),
            "transferred_out_prev": to_prev,
            "transferred_out_month": to_mo,
            "transferred_out_cumulative": add_triplets(to_prev, to_mo),
            "transferred_in_prev": ti_prev,
            "transferred_in_month": ti_mo,
            "transferred_in_cumulative": add_triplets(ti_prev, ti_mo),
        })

    grade_groups = []
    total_summary = empty_summary_row("grand-total", "TOTAL", "ALL")

    for g_label, sec_rows in groups_map.items():
        subtotal = empty_summary_row(f"subtotal-{g_label}", "SUBTOTAL", g_label)
        for row in sec_rows:
            for key in SUMMABLE_KEYS:
                subtotal[key] = add_triplets(subtotal[key], row[key])
        subtotal["attendance_percentage"] = compute_attendance_pct(
            subtotal["daily_average"], subtotal["registered_end"]
        )
        grade_groups.append({
            "grade_level": g_label,
            "sections": sec_rows,
            "subtotal": subtotal,
        })
        for key in SUMMABLE_KEYS:
            total_summary[key] = add_triplets(total_summary[key], subtotal[key])

    total_summary["attendance_percentage"] = compute_attendance_pct(
        total_summary["daily_average"], total_summary["registered_end"]
    )

    active_sy = target_sy or SchoolYear.objects.filter(is_active=True).first()
    resolved_sy = active_sy.code if active_sy else (str(school_year).strip() or f"{year}-{year + 1}")

    return {
        **get_school_metadata(request),
        "school_year": resolved_sy,
        "month": clean_month,
        "year": year,
        "grade_groups": grade_groups,
        "non_graded_summary": None,
        "total_summary": total_summary,
    }

def _parse_int_param(raw, *, default: int, name: str, min_val: int, max_val: int) -> int:
    if raw in (None, ""):
        return default
    try:
        val = int(str(raw).strip())
    except (TypeError, ValueError):
        raise AppError(f"INVALID_{name.upper()}", f"{name} must be a valid integer.", 400)
    if not (min_val <= val <= max_val):
        raise AppError(f"INVALID_{name.upper()}", f"{name} must be between {min_val} and {max_val}.", 400)
    return val


def _parse_month_int(raw, default: int) -> int:
    if raw in (None, ""):
        return default
    num, _ = resolve_month(raw)
    return num


def _authorize_section_access(user, section_id: int) -> bool:
    if is_school_leader(user):
        return True
    teacher = resolve_teacher_for_user(user)
    return bool(teacher and section_id in get_teacher_section_ids(teacher))


def _require_staff(user) -> None:
    if get_user_role(user) == "PARENT" and not is_school_leader(user):
        raise AppError("FORBIDDEN", "Staff or school leadership access required.", 403)

        raw_tid = pk or request.query_params.get("faculty_id") or request.query_params.get("teacher_id")

        list_only = str(request.query_params.get("list", "")).lower() in ("1", "true", "yes")
        return Response(
            get_monthly_dtr_data(
                user=request.user,
                month=month,
                year=year,
                teacher_id=t_id,
                list_only=list_only,
                request=request,
            )
        )

    def post(self, request, pk: int | None = None):
        from .services import process_monthly_dtr_action

        _require_staff(request.user)
        month = _parse_month_int(request.data.get("month"), timezone.localdate().month)
        year = _parse_int_param(
            request.data.get("year"),
            default=timezone.localdate().year,
            name="year",
            min_val=1900,
            max_val=2100,
        )
        raw_tid = pk or request.data.get("faculty_id") or request.data.get("teacher_id")
        t_id = int(raw_tid) if (raw_tid and str(raw_tid).isdigit()) else None

        result = process_monthly_dtr_action(
            request=request,
            action=str(request.data.get("action") or ""),
            month=month,
            year=year,
            teacher_id=t_id,
            data=request.data,
        )
        return Response(result, status=status.HTTP_200_OK)


class ReportExportView(APIView):
    """Generates printable HTML exports for SF1, SF2, SF4, and DTR reports (used by mobile & web)."""

    permission_classes = [permissions.IsAuthenticated]

    def _handle(self, request, params):
        _require_staff(request.user)
        report_type = str(params.get("report_type") or params.get("type") or "SF1").strip().upper()
        year = _parse_int_param(
            params.get("year"),
            default=timezone.localdate().year,
            name="year",
            min_val=1900,
            max_val=2100,
        )
        raw_sec = params.get("section_id")
        sec_id = int(raw_sec) if (raw_sec and str(raw_sec).isdigit()) else None

        if report_type == "SF1":
            if not sec_id:
                raise AppError("SECTION_REQUIRED", "Valid section_id is required for SF1 export.", 400)
            if not _authorize_section_access(request.user, sec_id):
                raise AppError("FORBIDDEN", "Access denied to this section.", 403)
            context = generate_sf1_report(section_id=sec_id, school_year=params.get("school_year"), request=request)
            return render(request, "reports/sf1.html", context)

        if report_type == "SF2":
            if not sec_id:
                raise AppError("SECTION_REQUIRED", "Valid section_id is required for SF2 export.", 400)
            if not _authorize_section_access(request.user, sec_id):
                raise AppError("FORBIDDEN", "Access denied to this section.", 403)
            data = generate_sf2_report(
                section_id=sec_id,
                month_name=params.get("month") or "October",
                year=year,
                request=request,
            )
            rows_html = "".join(
                f"<tr><td>{escape(r['lrn'])}</td><td>{escape(r['name'])}</td><td>{escape(r['sex'])}</td>"
                f"<td>{r['total_absent']}</td><td>{r['total_tardy']}</td></tr>"
                for r in (data["males"] + data["females"])
            )
            html = (
                f"<!doctype html><html><head><meta charset='utf-8'><title>SF2 - {escape(data['section_name'])}</title>"
                "<style>body{font-family:Arial,sans-serif;font-size:12px}table{width:100%;border-collapse:collapse}"
                "th,td{border:1px solid #333;padding:4px 6px;text-align:left}</style></head><body>"
                f"<h2>School Form 2 (SF2) — {escape(data['school_name'])}</h2>"
                f"<p>Section: <b>{escape(data['grade_level'])} - {escape(data['section_name'])}</b> | "
                f"Month: <b>{escape(data['month'])} {data['year']}</b> | Adviser: <b>{escape(data['adviser_name'])}</b></p>"
                "<table><thead><tr><th>LRN</th><th>Learner Name</th><th>Sex</th><th>Total Absent</th><th>Total Tardy</th></tr></thead>"
                f"<tbody>{rows_html}</tbody></table></body></html>"
            )
            return HttpResponse(html, content_type="text/html; charset=utf-8")

        if report_type == "SF4":
            if not is_school_leader(request.user):
                raise AppError("FORBIDDEN", "Leadership access required for SF4 export.", 403)
            data = generate_sf4_report(
                month_name=params.get("month") or "October",
                year=year,
                school_year=params.get("school_year") or "",
                request=request,
            )
            rows_html = ""
            for grp in data.get("grade_groups", []):
                for sec in grp.get("sections", []):
                    rows_html += (
                        f"<tr><td>{escape(grp['grade_level'])}</td><td>{escape(sec['section_name'])}</td>"
                        f"<td>{escape(sec['adviser_name'])}</td><td>{sec['registered_end']['total']}</td>"
                        f"<td>{sec['daily_average']['total']}</td><td>{sec['attendance_percentage']['total']}%</td></tr>"
                    )
            html = (
                f"<!doctype html><html><head><meta charset='utf-8'><title>SF4 - {escape(data['month'])} {data['year']}</title>"
                "<style>body{font-family:Arial,sans-serif;font-size:12px}table{width:100%;border-collapse:collapse}"
                "th,td{border:1px solid #333;padding:4px 6px;text-align:left}</style></head><body>"
                f"<h2>School Form 4 (SF4) — {escape(data['school_name'])}</h2>"
                f"<p>Month: <b>{escape(data['month'])} {data['year']}</b></p>"
                "<table><thead><tr><th>Grade</th><th>Section</th><th>Adviser</th><th>Registered</th><th>ADA</th><th>% Attendance</th></tr></thead>"
                f"<tbody>{rows_html}</tbody></table></body></html>"
            )
            return HttpResponse(html, content_type="text/html; charset=utf-8")

        if report_type == "DTR":
            month_num = _parse_month_int(params.get("month"), timezone.localdate().month)
            raw_tid = params.get("faculty_id") or params.get("teacher_id")
            t_id = int(raw_tid) if (raw_tid and str(raw_tid).isdigit()) else None
            data = get_monthly_dtr_data(user=request.user, month=month_num, year=year, teacher_id=t_id, request=request)
            rows_html = "".join(
                f"<tr><td>{r['day']}</td><td>{escape(r['am_arrival'])}</td><td>{escape(r['am_departure'])}</td>"
                f"<td>{escape(r['pm_arrival'])}</td><td>{escape(r['pm_departure'])}</td><td>{escape(r['status'])}</td></tr>"
                for r in data["rows"]
                if r["status"] != "OUT_OF_MONTH"
            )
            html = (
                f"<!doctype html><html><head><meta charset='utf-8'><title>CSC Form 48 - {escape(data['faculty_name'])}</title>"
                "<style>body{font-family:Arial,sans-serif;font-size:12px}table{width:100%;border-collapse:collapse}"
                "th,td{border:1px solid #333;padding:4px 6px;text-align:center}</style></head><body>"
                f"<h2>Civil Service Form No. 48 (DTR) — {escape(data['faculty_name'])}</h2>"
                f"<p>Month: <b>{escape(data['month'])} {data['year']}</b> | Days Present: <b>{data['days_present']}</b></p>"
                "<table><thead><tr><th>Day</th><th>AM In</th><th>AM Out</th><th>PM In</th><th>PM Out</th><th>Status</th></tr></thead>"
                f"<tbody>{rows_html}</tbody></table></body></html>"
            )
            return HttpResponse(html, content_type="text/html; charset=utf-8")

        raise AppError("INVALID_REPORT_TYPE", f"Unsupported report type '{report_type}'.", 400)

    def get(self, request):
        return self._handle(request, request.query_params)

    def post(self, request):
        return self._handle(request, request.data)


class ReportAuditLogView(APIView):
    """Records verifiable audit entries whenever SF1, SF2, SF4, or DTR forms are printed or exported."""
    permission_classes = [permissions.IsAuthenticated]

    def post(self, request):
        _require_staff(request.user)
        raw_section = request.data.get("section_id")
        record_id = int(raw_section) if (raw_section and str(raw_section).isdigit()) else 0
        action_code = str(request.data.get("action") or "EXPORT").strip().upper()[:10]
        report_name = str(request.data.get("report_name") or "School Form").strip()[:80]
        tracking_id = str(request.data.get("document_tracking_id") or "").strip()[:80]

        record_audit_log(
            request=request,
            table_name="deped_reports",
            record_id=record_id,
            action=action_code,
            new_data={"report_name": report_name, "tracking_id": tracking_id},
        )
        return Response({"ok": True, "logged": True}, status=status.HTTP_201_CREATED)

class SystemStatusView(APIView):
    permission_classes = [permissions.IsAuthenticated]

    def get(self, request):
        _require_staff(request.user)
        return Response(get_system_status())

    def post(self, request):
        if not is_school_leader(request.user):
            raise AppError("FORBIDDEN", "School leadership access required.", 403)
        action_name = str(request.data.get("action") or "").strip().upper()
        retried = 0
        if action_name == "RETRY_FAILED":
            retried = retry_pending_push_notifications()
        data = dict(get_system_status())
        data["retried_count"] = retried
        return Response(data)
